\section{Methods}
\label{sec:methods}
%==============================================================================

Two groups of methods are developed. \emph{Intrinsic exploration bonuses} (count-based,
RND, elliptical) define $r^{\mathrm{int}}$ added to the SAC reward on every replay
sample as $r = r^{\mathrm{ext}} + \beta\,r^{\mathrm{int}}$. \emph{Distance-to-ground-truth
metrics} score how close a method's bonus field $p$ (evaluated on the full grid $G$) is
to the oracle field $g$ of \texttt{gt\_position\_velocity}.

\label{sec:method-table}

these bonus fields are cataloged separately in Table~\ref{tab:metrics}
(Section~\ref{sec:metrics}). The per-output RND distance is written $d(x)$ and defined in
Section~\ref{sec:rnd}.

\subsection{Intrinsic bonus methods}
\label{sec:method-descriptions}

\subsubsection{Baseline: no exploration}
\label{sec:no-exploration}

\paragraph{Category.}
The control condition: no intrinsic bonus. The replay buffer returns the unmodified
environment reward, so $r = r^{\mathrm{ext}}$ and $\beta$ is forced to $0$ in code.

\paragraph{\texttt{no\_exploration}.}
Plain SAC with extrinsic reward only:
\begin{align}
r^{\mathrm{int}} = 0.
\end{align}

\subsubsection{Count-based bonuses (oracle)}
\label{sec:count-based}

\paragraph{Category.}
Discretize the state to a grid cell, count how often each cell has been visited, and
make the bonus shrink with the count. The common formula is
$r^{\mathrm{int}}(n) = \min(1,\, n^{\alpha})$ for $n>0$, and $r^{\mathrm{int}}=1$ for $n\le0$ (unvisited cell gets the
maximum bonus). The default exponent $\alpha=-0.5$ gives $r^{\mathrm{int}} = \min(1,\,1/\sqrt{n})$.
Counts are maintained by the environment wrapper during stepping; the model's
\texttt{update()} is a no-op. These bonuses use the (oracle) true visit statistics, so
they serve as the reference against which learned bonuses are compared.

\paragraph{\texttt{gt\_position}.}
$n$ is the visit count of the position cell containing the next state, on the $9\times12$
position grid.
\begin{align}
r^{\mathrm{int}} = \min(1,\, n^{\alpha}).
\end{align}

\paragraph{\texttt{gt\_position\_velocity}.}
$n$ is the visit count of the joint (position, velocity) cell, on the
$9\times12\times10\times10$ grid (10 velocity bins per axis).
\begin{align}
r^{\mathrm{int}} = \min(1,\, n^{\alpha}).
\end{align}
The bonus field of this method over the full grid $G$ is the \emph{ground-truth field}
$g$ used by every distance metric.

\subsubsection{Random Network Distillation bonuses}
\label{sec:rnd}

\paragraph{Category.}
A frozen random target network $f_\theta$ and a trainable predictor $g_\phi$ both map an
input feature $x$ to $\mathbb{R}^m$ ($m=128$). The input is first standardized,
$\tilde{x} = \mathrm{clip}\!\big((x-\mu)/\sqrt{\sigma^2+10^{-8}},\,-5,\,5\big)$, using a
running mean/variance updated on the raw input. The per-output distance is
\begin{align}
d(x) \;=\;
\begin{cases}
\tfrac{1}{2}\sum_{k=1}^{m}\big(f_\theta(\tilde{x})_k - g_\phi(\tilde{x})_k\big)^2 & \text{(mse, default)}\\[4pt]
\sum_{k=1}^{m}\big|f_\theta(\tilde{x})_k - g_\phi(\tilde{x})_k\big| & \text{(abs).}
\end{cases}
\end{align}
The predictor is trained to minimize the mean of $d$ over the batch; the bonus is $d$
evaluated on the current sample. Variants differ only in the input feature $x$.

\paragraph{\texttt{rnd\_next\_state}.}
$x = s'$ (the full next state).
\begin{align}
r^{\mathrm{int}} = d(s').
\end{align}

\paragraph{\texttt{rnd\_next\_state\_position\_only}.}
$x = s'_{1:2}$ (the first two coordinates of the next state, i.e.\ position only).
\begin{align}
r^{\mathrm{int}} = d(s'_{1:2}).
\end{align}

\paragraph{\texttt{rnd\_state}.}
$x = s$ (the current state).
\begin{align}
r^{\mathrm{int}} = d(s).
\end{align}

\paragraph{\texttt{rnd\_state\_action}.}
$x = [\,s\,;a\,]$ (state concatenated with action); action-conditioned, so no
distance-to-GT vector is logged.
\begin{align}
r^{\mathrm{int}} = d([\,s\,;a\,]).
\end{align}

\paragraph{\texttt{rnd\_state\_action\_next\_state}.}
$x = [\,s\,;a\,;s'\,]$ (state, action, next state); action-conditioned, no
distance-to-GT vector.
\begin{align}
r^{\mathrm{int}} = d([\,s\,;a\,;s'\,]).
\end{align}

\paragraph{\texttt{rnd\_linear\_next\_state} (linear RND).}
Both networks first apply one\emph{shared, frozen} random trunk---a fixed feature map $\varphi=\mathrm{ReLU}(W_1\tilde{x}+b_1)\in\mathbb{R}^{256}$ whose weights are copied from the target into the predictor and held fixed. The target then applies a frozen random linear head, $f_\theta=W_\theta\,\varphi$; the predictor's linear head $g_\phi=W_\phi\,\varphi$ is the \emph{only} trained part, fit by least squares (the same squared-error objective the other RND variants use) to match the random target over visited inputs---i.e.\ a linear regression onto $f_\theta$ in the fixed feature space $\varphi$. The input is $x=s'$. The bonus is the true
Euclidean norm
\begin{align}
r^{\mathrm{int}} = \lVert g_\phi(\tilde{x}) - f_\theta(\tilde{x})\rVert_2
= \sqrt{\max\Big(\textstyle\sum_{k=1}^{m}(g_\phi(\tilde{x})_k - f_\theta(\tilde{x})_k)^2,\,10^{-8}\Big)}.
\end{align}
\smallskip\noindent\textbf{Why the Euclidean norm, not the shared $d$?} The other RND variants read out the squared-error distance $d$ of Section~\ref{sec:rnd} as the bonus; linear RND reads out the true norm $\lVert e\rVert_2$ (with $e=g_\phi(\tilde{x})-f_\theta(\tilde{x})$) instead, and the difference is deliberate. The count oracle decays as $1/\sqrt{n}$---a standard error (Section~\ref{sec:opt-defs}), i.e.\ a standard-deviation--scale quantity. The norm $\lVert e\rVert_2$ lives on that scale and \emph{can} decay as $1/\sqrt{n}$, whereas the squared distance $d=\tfrac12\lVert e\rVert_2^2$ is variance-scale and decays as $1/n$. The linear head sharpens the link: fitting a linear map onto the fixed feature $\varphi$ gives a converged residual $\propto\sqrt{\varphi^\top\Lambda^{-1}\varphi}$, the elliptical/Mahalanobis bonus (Section~\ref{sec:elliptical}), under the cumulative-ridge idealization of Section~\ref{sec:opt-defs}. So the norm---not $d$---is the read-out that keeps this variant on the same $1/\sqrt{n}$ scale as the count oracle and the elliptical bonus; reusing $d(x)$ would put it back on the $1/n$ scale and break that correspondence. Two caveats, both from Section~\ref{sec:opt-defs}: only the bonus read-out changes---the training loss is the same squared error as the other variants; and as shipped the predictor head is trained with constant-rate Adam, which accumulates no count, so the realized bonus saturates rather than actually reaching $1/\sqrt{n}$ (the norm read-out is necessary for that scaling, not by itself sufficient).

\paragraph{RND ensemble ($N>1$).}
$N$ independent predictors share one target, with per-predictor distances $d_q$. The
bonus adds the disagreement spread:
\begin{align}
r^{\mathrm{int}} &= \bar{d} + \beta_{\mathrm{std}}\,s_d, \\
\bar{d} &= \tfrac1N\sum_{q=1}^N d_q, \qquad
s_d = \sqrt{\tfrac{1}{N-1}\sum_{q=1}^N (d_q-\bar{d})^2}.
\end{align}
With the default $N=1$, $s_d\equiv0$ and the bonus is the single distance.

\subsubsection{Elliptical (Mahalanobis) bonus}
\label{sec:elliptical}

\paragraph{What problem this bonus is trying to solve.}
The count bonuses in Section~\ref{sec:count-based} are easy to understand because
they use an explicit visit count $n$: a cell that has been visited many times gets
a small bonus, and a cell that has been visited rarely gets a large bonus. The
problem is that exact counting is only easy after we discretize the world into
cells. For a continuous input such as $(s,a)$, exact counting is no longer natural:
two states can be different as real vectors but still represent almost the same
place in the maze, or almost the same behavior.

The elliptical bonus is a continuous analogue of counting. Instead of asking,
``How many times did we visit exactly this cell?'', it asks,
``How much data do we have in the feature-space direction of this $(s,a)$ pair?''
The answer is computed with a fixed random feature encoder
\[
\varphi(s,a)\in\mathbb{R}^{d_\varphi}.
\]
A direction in feature space that has appeared often should have low uncertainty
and therefore a small bonus. A direction that has appeared rarely should have high
uncertainty and therefore a large bonus.

\paragraph{Why it is called RND and why it is called elliptical.}
The method name \texttt{rnd\_elliptical} combines two ideas.

First, the ``RND'' part means that the feature map is random and frozen, just as
the target network $f_\theta$ in ordinary RND is random and frozen. However, the
elliptical bonus is \emph{not} an ordinary RND prediction-error bonus: there is no
trainable predictor $g_\phi$ whose error is directly used as the bonus. Instead,
the random network produces fixed features $\varphi(s,a)$, and the bonus is
computed analytically from the feature covariance matrix $\Lambda$.

Second, the ``elliptical'' part comes from linear regression and linear-bandit
confidence bounds. A covariance or precision matrix defines ellipses in two
dimensions and ellipsoids in higher dimensions. In this method, the matrix
$\Lambda$ defines which feature-space directions are well covered by data and
which directions are poorly covered. The quantity
\[
\varphi(s,a)^\top \Lambda^{-1}\varphi(s,a)
\]
is the squared length of the feature vector measured in the geometry defined by
$\Lambda^{-1}$. This is a Mahalanobis-type norm. The bonus is the square root of
that squared length:
\begin{align}
r^{\mathrm{int}}(s,a)
=
\sqrt{
\max\Big(
\varphi(s,a)^\top\Lambda^{-1}\varphi(s,a),
10^{-8}
\Big)
}.
\end{align}

\paragraph{The input and the frozen feature encoder.}
For the action-conditioned version used by \texttt{rnd\_elliptical}, the input is
the concatenation of state and action:
\[
[s;a]\in\mathbb{R}^{4+2}=\mathbb{R}^6.
\]
The state $s=(x,y,v_x,v_y)$ has four coordinates, and the action
$a\in[-1,1]^2$ has two coordinates. The frozen random encoder maps this
six-dimensional vector into a $d_\varphi$-dimensional feature vector:
\begin{align}
\varphi(s,a)
=
W_2\,\mathrm{ReLU}\!\big(W_1[s;a]+b_1\big)
\in\mathbb{R}^{d_\varphi},
\qquad d_\varphi=128.
\end{align}
Here $W_1$, $b_1$, and $W_2$ are sampled at initialization and then held fixed.
Thus, $\varphi$ is not trained. It is a random, fixed coordinate system in which
we measure novelty.

\paragraph{What ``stacking'' means.}
Suppose a replay-buffer minibatch contains $b$ transitions
\[
\{(s_i,a_i)\}_{i=1}^{b}.
\]
For each transition, compute the feature vector
\[
\varphi_i=\varphi(s_i,a_i)\in\mathbb{R}^{d_\varphi}.
\]
A single feature vector $\varphi_i$ is best viewed as a column vector:
\[
\varphi_i
=
\begin{bmatrix}
\varphi_{i1}\\
\varphi_{i2}\\
\vdots\\
\varphi_{i d_\varphi}
\end{bmatrix}
\in\mathbb{R}^{d_\varphi}.
\]
To build a matrix out of the whole batch, we put one feature vector per row:
\begin{align}
\Phi_B
=
\begin{bmatrix}
\varphi_1^\top\\
\varphi_2^\top\\
\vdots\\
\varphi_b^\top
\end{bmatrix}
=
\begin{bmatrix}
\varphi_{11} & \varphi_{12} & \cdots & \varphi_{1d_\varphi}\\
\varphi_{21} & \varphi_{22} & \cdots & \varphi_{2d_\varphi}\\
\vdots       & \vdots       & \ddots & \vdots\\
\varphi_{b1} & \varphi_{b2} & \cdots & \varphi_{bd_\varphi}
\end{bmatrix}
\in\mathbb{R}^{b\times d_\varphi}.
\end{align}
This is the meaning of ``stacking'': the batch dimension becomes the row
dimension, and the feature dimension becomes the column dimension. Rows are
samples; columns are random features.

\paragraph{What the matrix multiplication $\Phi_B^\top\Phi_B$ means.}
The product $\Phi_B^\top\Phi_B$ has shape
\[
\Phi_B^\top\Phi_B
\in
\mathbb{R}^{d_\varphi\times d_\varphi}.
\]
It is not a mysterious operation. Its $(j,k)$ entry is
\begin{align}
\big[\Phi_B^\top\Phi_B\big]_{jk}
=
\sum_{i=1}^{b}\varphi_{ij}\varphi_{ik}.
\end{align}
Thus:
\begin{itemize}[leftmargin=1.4em,itemsep=2pt,topsep=2pt]
\item the diagonal entry $(j,j)$ is
\[
\sum_{i=1}^{b}\varphi_{ij}^2,
\]
which says how strongly feature $j$ is activated in the batch;
\item the off-diagonal entry $(j,k)$ is
\[
\sum_{i=1}^{b}\varphi_{ij}\varphi_{ik},
\]
which says whether features $j$ and $k$ tend to be active together.
\end{itemize}
So $\Phi_B^\top\Phi_B$ summarizes which feature directions have been observed in
the batch. It turns a list of $b$ feature vectors into one
$d_\varphi\times d_\varphi$ matrix describing feature coverage.

The current implementation uses the per-batch matrix
\begin{align}
\Lambda_B
=
\frac{1}{b}\Phi_B^\top\Phi_B+\lambda I
\in\mathbb{R}^{d_\varphi\times d_\varphi}.
\end{align}
The regularizer $\lambda I$ is added so that the matrix is invertible. Without
this term, some feature directions might have zero variance in the batch, making
the inverse undefined.

\paragraph{What $\Lambda^{-1}\varphi$ means.}
The expression
\[
\varphi(s,a)^\top\Lambda^{-1}\varphi(s,a)
\]
can be read in three steps. Let
\[
\varphi=\varphi(s,a).
\]
First solve the linear system
\begin{align}
\Lambda u=\varphi.
\end{align}
The solution is
\[
u=\Lambda^{-1}\varphi.
\]
Second, take the dot product between the original feature vector and the rescaled
feature vector:
\begin{align}
q
=
\varphi^\top u
=
\varphi^\top\Lambda^{-1}\varphi.
\end{align}
Third, take the square root:
\begin{align}
r^{\mathrm{int}}(s,a)=\sqrt{\max(q,10^{-8})}.
\end{align}
In code, it is usually better not to explicitly form $\Lambda^{-1}$. A more stable
implementation solves $\Lambda u=\varphi$ directly, or uses a Cholesky factor
$LL^\top=\Lambda$ and computes
\begin{align}
\varphi^\top\Lambda^{-1}\varphi
=
\lVert L^{-1}\varphi\rVert_2^2.
\end{align}

\paragraph{The geometric meaning.}
Diagonalize the positive-definite matrix $\Lambda$ as
\[
\Lambda=\sum_{j=1}^{d_\varphi}\ell_j v_jv_j^\top,
\qquad
\ell_j>0,
\]
where $v_j$ are orthonormal directions in feature space and $\ell_j$ are the
amounts of data coverage in those directions. Then
\begin{align}
\varphi^\top\Lambda^{-1}\varphi
=
\sum_{j=1}^{d_\varphi}
\frac{(v_j^\top\varphi)^2}{\ell_j}.
\end{align}
This equation explains the whole method. If the current feature vector points in
a direction with large $\ell_j$, then that direction has been well covered by the
batch or by past data, so the contribution to the bonus is small. If the current
feature vector points in a direction with small $\ell_j$, then that direction is
poorly covered, so the contribution to the bonus is large.

Thus the elliptical bonus is high when $(s,a)$ points toward an under-sampled
feature direction, and low when $(s,a)$ points toward a well-sampled feature
direction.

\paragraph{Connection to ordinary counts.}
The count bonus is recovered as a special case. Suppose the feature vector is a
one-hot cell indicator. That is, if the current transition belongs to cell $j$,
then
\[
\varphi(s,a)=e_j,
\]
where $e_j$ is the vector with a $1$ in coordinate $j$ and $0$ elsewhere. If we
use the cumulative matrix
\[
\Lambda_t
=
\lambda I+\sum_{\tau\le t}\varphi(s_\tau,a_\tau)\varphi(s_\tau,a_\tau)^\top,
\]
then
\[
\Lambda_t
=
\operatorname{diag}(\lambda+n_1,\lambda+n_2,\ldots),
\]
where $n_j$ is the number of visits to cell $j$. For a transition in cell $j$,
\begin{align}
\varphi(s,a)^\top\Lambda_t^{-1}\varphi(s,a)
=
e_j^\top\Lambda_t^{-1}e_j
=
\frac{1}{\lambda+n_j}.
\end{align}
Therefore
\begin{align}
r^{\mathrm{int}}(s,a)
=
\frac{1}{\sqrt{\lambda+n_j}}
\approx
\frac{1}{\sqrt{n_j}}.
\end{align}
This is exactly the same decay law as the count oracle in
Section~\ref{sec:count-based}. The elliptical bonus is therefore a generalized
count: it behaves like a count when features are one-hot, and like a soft
feature-space count when features are continuous random vectors.

\paragraph{The elliptical family: one bonus, a covariance rule, and an update timing.}
Every elliptical method shares the same frozen feature $\varphi$, the same feature normalization, the
same quadratic-form solve, and the same bonus
\begin{align}
r^{\mathrm{int}}(s,a)
=
\min\!\Big(
\sqrt{\max\big(\varphi(s,a)^\top\Lambda^{-1}\varphi(s,a),\,\epsilon_q\big)},\;
r_{\max}^{\mathcal{E}}
\Big),
\end{align}
clipped to $r_{\max}^{\mathcal{E}}=5$ by default. (With a small ridge an under-covered feature
direction can give a very large raw bonus; the clip bounds the rare reward spikes that would otherwise
destabilize SAC. Setting $r_{\max}^{\mathcal{E}}=\infty$ recovers the unclipped value.) The members
differ along two independent choices: \emph{which covariance rule} maintains $\Lambda$, and \emph{when}
that rule fires. Two covariance rules are implemented.

\emph{Batch elliptical} (the shipped \texttt{rnd\_elliptical}) replaces $\Lambda$ with the covariance
of the current update batch $U$ of size $n$:
\begin{align}
\Lambda=\frac{1}{n}\Phi_U^\top\Phi_U+\lambda I.
\end{align}
This is a within-batch novelty score; it does not remember earlier visits, so it is not a lifetime
count.

\emph{Global elliptical} (\texttt{rnd\_elliptical\_global}) keeps a true running total of feature
coverage. It is maintained \textbf{iteratively}, by one running $d_\varphi\times d_\varphi$ matrix that
each update adds the new batch's outer products to:
\begin{align}
\Lambda_t=\Lambda_{t-1}+\Phi_{U_t}^\top\Phi_{U_t},
\qquad
\Lambda_0=\lambda I.
\end{align}
This is the \emph{only} way the global covariance is computed: the closed form
$\Lambda_t=\lambda I+\sum_{\tau\le t}\varphi_\tau\varphi_\tau^\top$ is merely the value this recurrence
equals --- it is never recomputed from the stored history. The update costs $O(|U_t|\,d_\varphi^2)$
time plus the $O(d_\varphi^3)$ Cholesky refactor, and $O(d_\varphi^2)$ memory (with $d_\varphi=128$,
only $128^2=16384$ entries), \emph{not} storage of the replay buffer. The ridge default is
$\lambda=10^{-2}$ for both rules (interpretable because unit-norm features have $\lVert\varphi\rVert\le
1$). A discounted variant
$\Lambda_t=\rho_\Lambda\Lambda_{t-1}+\Phi_{U_t}^\top\Phi_{U_t}+(1-\rho_\Lambda)\lambda I$ that down-weights
old visits is a possible future extension and is not part of the first run.

\paragraph{When statistics are updated: on sample (default) versus on add.}
What the batch above calls $U_t$ --- and therefore what every intrinsic statistic counts --- is decided
by the replay buffer, which is the single switch between two timings. The buffer's \texttt{add()} stores
only the extrinsic reward and calls the model's \texttt{observe()}; its \texttt{sample()} computes the
bonus and calls the model's \texttt{update()} once per SAC gradient step on the replay minibatch:
\begin{lstlisting}[style=pythoncode]
# VectorIntrinsicReplayBuffer
def add(self, obs, next_obs, action, reward, done, infos):
    super().add(obs, next_obs, action, reward, done, infos)   # stores the extrinsic reward only
    if self.intrinsic_reward_model is not None and self.beta > 0.0:
        new = {"observations": obs, "next_observations": next_obs, "actions": action}
        self.intrinsic_reward_model.observe(new)              # fires only for update_timing="add"

def sample(self, batch_size, env=None):
    batch = super()._get_samples(batch_inds, env)             # raw stored transitions (no obs-rms)
    intrinsic = self.intrinsic_reward_model.compute(samples)  # the bonus, on the sampled batch
    self.intrinsic_reward_model.update(samples)               # fires only for update_timing="sample"
    return batch_with_reward(extrinsic + self.beta * intrinsic)
\end{lstlisting}

\emph{What RND does (on sample), and how it differs in input handling.} For \texttt{rnd\_next\_state}
the bonus is the prediction error on the next state $s'$. RND's running observation mean/variance and
its predictor distillation step both happen inside \texttt{update()}, i.e.\ \textbf{on sample} (from
replay draws), and RND \emph{standardizes its input} with that running mean/variance:
\begin{lstlisting}[style=pythoncode]
# RND.update  (called by buffer.sample())
def update(self, samples):
    x = self._get_feature_tensor(samples)            # rnd_next_state -> next state s'
    if self.use_obs_norm:
        self.obs_rms.update(x.detach().cpu().numpy())  # running mean/var, on sample
    x = self._normalize_obs(x)                       # standardize, then clip to [-5, 5]
    src, tgt = self.predictor(x), self.target(x).detach()
    loss = ((src - tgt) ** 2).mean()                 # distillation step, on sample
    loss.backward(); self.opt.step()
\end{lstlisting}
The elliptical family is deliberately different on the input side: its encoder consumes the \textbf{raw}
state $s$ and action $a$. It applies \emph{no} observation RMS; the only normalization is the in-family
feature normalization $\varphi=z/\lVert z\rVert$ (Section ``Raw features and normalized features''). So
the elliptical bonus never standardizes the state upstream of the encoder.

\emph{What batch elliptical did, and what global elliptical does.} In the elliptical family the update
timing is an explicit field \texttt{update\_timing} that is orthogonal to the covariance rule, so
\texttt{update()} (sample) and \texttt{observe()} (add) both route to one \texttt{\_update\_covariance};
the \textbf{default is sample timing} for every member. Batch elliptical replaces $\Lambda$ from the
sampled minibatch; global elliptical accumulates onto its running $\Lambda$:
\begin{lstlisting}[style=pythoncode]
# EllipticalBonus (family base): timing is orthogonal to the covariance rule
def update(self, samples):                  # called by buffer.sample()
    if self.update_timing == "sample":      # default for the whole family
        self._update_covariance(samples)

def observe(self, samples):                 # called by buffer.add()
    if self.update_timing == "add":
        self._update_covariance(samples)

def _update_covariance(self, samples):      # BATCH rule: replace Lambda from this batch
    phi = self._features_for_update(samples)
    self._cov = (phi.T @ phi) / len(phi) + self.regularization * self._eye
    self._prepare_solver()                  # refactor the Cholesky every update

class GlobalEllipticalBonus(EllipticalBonus):
    def _update_covariance(self, samples):  # GLOBAL rule: accumulate (Lambda_0 = lambda I)
        phi = self._features_for_update(samples)
        self._cov = self._cov + phi.T @ phi  # one running matrix, never recomputed from history
        self._prepare_solver()
\end{lstlisting}
The two timings count different things. On the \textbf{default sample timing}, $\Lambda$ accumulates the
replay minibatches, so a frequently re-sampled transition contributes more than once --- the global
covariance then tracks replay-draw frequency. On \textbf{add timing} (\texttt{update\_timing="add"}),
each freshly collected transition enters $\Lambda$ exactly once through \texttt{observe()}, so the
covariance is a true environment-visitation count (the count-oracle reading of the previous paragraph).
Add timing is available to the whole family; sample timing is the default.

\paragraph{No Cholesky-refresh cadence.} The Cholesky factor is recomputed on every covariance update,
with no caching cadence. Nothing is saved by amortizing it: batch elliptical already pays a full
$O(d_\varphi^3)$ factorization on every sampled minibatch, so refactoring the global covariance once per
update --- the same cost --- has nothing to amortize against.

\subsection{Optimizers from first principles}
\label{sec:opt-defs}

\paragraph{Why optimizers matter for exploration bonuses.}
For ordinary RND, the intrinsic bonus is the prediction error between a trainable
predictor $g_\phi$ and a frozen random target $f_\theta$:
\[
e_\phi(\tilde{x})
=
g_\phi(\tilde{x})-f_\theta(\tilde{x}).
\]
The target network $f_\theta$ never changes. Only the predictor parameters
$\phi$ are trained. Therefore, the speed at which the bonus decreases at a
visited input is controlled by the optimizer used to update $\phi$.

If the optimizer makes the predictor forget old visits, then the RND bonus will
not behave like a count. If the optimizer accumulates information from all visits,
then the RND bonus can become closer to the count oracle's $1/\sqrt{n}$ decay.


Here ``bonus'' must be separated from ``loss.'' The predictor optimizer updates
$\phi$ by minimizing a supervised regression loss against the frozen random target.
SAC does not optimize this predictor loss directly; SAC receives an intrinsic
reward readout computed from the predictor--target residual. Thus the RND
predictor objective and the intrinsic-bonus readout can be the same scalar in the
default implementation, but they do not have to be the same scalar.


\begin{definition}[Optimizer]
An optimizer is an update rule for the trainable parameters $\phi$. At update
step $t$, the optimizer receives a minibatch loss $L_t(\phi)$ and its gradient
\[
h_t=\nabla_\phi L_t(\phi_t).
\]
The optimizer then produces the next parameters $\phi_{t+1}$. In the simplest
case, the update has the form
\[
\phi_{t+1}=\phi_t-\eta h_t,
\]
where $\eta>0$ is the learning rate.

For the RND predictor, this loss is the predictor-training objective. It is not
the SAC return objective and it is not necessarily the same scalar that is passed
to SAC as the intrinsic bonus.

\end{definition}


\begin{definition}[Predictor objective and bonus readout]
Let
\[
e_\phi(\tilde{x})=g_\phi(\tilde{x})-f_\theta(\tilde{x})
\]
be the RND residual. The predictor optimizer minimizes a loss
\[
L_t(\phi)
\]
with respect to $\phi$. SAC receives a reward
\[
r_t^{\mathrm{train}}
=
r_t^{\mathrm{ext}}
+
\beta\,\mathcal{B}_\psi(x_t;\phi_t),
\]
where $\mathcal{B}_\psi$ is the chosen intrinsic-bonus readout. Therefore the
full system contains two different optimization problems:
\[
\boxed{
\text{RND predictor: minimize } L_t(\phi)
}
\qquad
\boxed{
\text{SAC policy: maximize return under }
r^{\mathrm{ext}}+\beta\mathcal{B}_\psi
}.
\]
The bonus readout affects SAC through the reward, but it is not itself the loss
that the RND optimizer differentiates.
\end{definition}


\paragraph{The RND predictor loss.}
For a minibatch of $b$ RND inputs $\{x_i\}_{i=1}^{b}$, where $x_i$ may be
$s_i$, $s_i'$, $[s_i;a_i]$, or $[s_i;a_i;s_i']$ depending on the RND variant, the
predictor is trained to minimize
\begin{align}
L_t(\phi)
=
\frac1b\sum_{i=1}^{b}
\frac12
\left\lVert
g_\phi(\tilde{x}_i)-f_\theta(\tilde{x}_i)
\right\rVert_2^2.
\end{align}
The gradient $h_t=\nabla_\phi L_t(\phi_t)$ tells us how to change $\phi$ to make
the predictor closer to the frozen target on the current minibatch.


This squared-error objective is the standard RND predictor objective: the original
RND construction trains the predictor against a fixed random target by minimizing
mean squared prediction error \citep{burda2019exploration}. For the experiments
in this document, define
\[
d_{\mathrm{mse}}(x;\phi)
=
\frac12
\left\lVert
g_\phi(\tilde{x})-f_\theta(\tilde{x})
\right\rVert_2^2.
\]
Then the predictor objective is
\[
L_t^{\mathrm{mse}}(\phi)
=
\frac1b\sum_{i=1}^{b}d_{\mathrm{mse}}(x_i;\phi).
\]
The canonical RND bonus readout is the same per-sample squared error:
\[
\mathcal{B}_{\mathrm{mse}}(x;\phi)
=
d_{\mathrm{mse}}(x;\phi).
\]
For the count-scale experiments, the recommended alternative readout is the L2
residual norm:
\[
\mathcal{B}_{\mathrm{l2}}(x;\phi)
=
\left\lVert
g_\phi(\tilde{x})-f_\theta(\tilde{x})
\right\rVert_2
=
\sqrt{2d_{\mathrm{mse}}(x;\phi)}.
\]
Thus the verified design choice is not ``always use MSE as the objective and L2
as the bonus in every RND experiment.'' The more precise recommendation is:
\[
\boxed{
\text{Keep } L_t^{\mathrm{mse}} \text{ as the predictor objective}
}
\qquad
\boxed{
\text{use } \mathcal{B}_{\mathrm{l2}} \text{ when the goal is to match a }
1/\sqrt{n}
\text{ count scale}
}.
\]
The squared-error bonus $\mathcal{B}_{\mathrm{mse}}$ should remain as the canonical
RND baseline, while $\mathcal{B}_{\mathrm{l2}}$ is the count-scale readout.



The reason is statistical scale. Squared error is variance-scale. Its square root
is standard-deviation or standard-error scale. Since the count oracle uses
$r^{\mathrm{int}}(n)=1/\sqrt{n}$, the count oracle is on standard-error scale, not
variance scale. Therefore, if the residual satisfies
\[
\mathbb{E}\lVert e_n\rVert_2^2=O(1/n),
\]
then
\[
\lVert e_n\rVert_2=O(1/\sqrt{n})
\]
has the same scale as the count oracle, while
\[
\tfrac12\lVert e_n\rVert_2^2=O(1/n)
\]
decays too quickly. This is also the scale used by linear confidence and
elliptical bonuses, where the uncertainty readout is a square root of a quadratic
form, such as
\[
\sqrt{\varphi^\top\Lambda^{-1}\varphi}
\]
rather than the quadratic form itself \citep{abbasi2011online}.


\paragraph{Gradient descent and stochastic gradient descent.}
The most basic optimizer is gradient descent:
\begin{align}
\phi_{t+1}
=
\phi_t-\eta h_t.
\end{align}
The gradient points in the direction of steepest increase of the loss, so the
optimizer subtracts the gradient to move downhill. In deep learning we usually use
a minibatch instead of the whole dataset; this is called stochastic gradient
descent, or SGD. The update formula is the same, but $h_t$ is a noisy minibatch
gradient.

For RND, constant-learning-rate SGD has one important weakness: it does not keep a
true count of how many times a state has been visited. It keeps moving with the
same step size $\eta$, so under minibatch noise the predictor error typically
falls quickly at first and then reaches a nonzero steady-state error.

\paragraph{Momentum.}
Momentum smooths the gradient by keeping a velocity-like variable:
\begin{align}
v_t &= \rho v_{t-1}+h_t,\\
\phi_{t+1} &= \phi_t-\eta v_t,
\qquad 0\le\rho<1.
\end{align}
Momentum is useful when consecutive gradients point in similar directions: it
accelerates motion in stable directions and damps small oscillations. However, if
$\eta$ is constant, momentum is still a constant-gain method. It improves
optimization speed, but it does not by itself create a count-like $1/\sqrt{n}$
bonus.

\paragraph{RMSProp.}
RMSProp rescales each parameter by a moving average of squared gradients:
\begin{align}
q_t
&=
\rho q_{t-1}+(1-\rho)(h_t\odot h_t),\\
\phi_{t+1}
&=
\phi_t
-
\eta\,
\frac{h_t}{\sqrt{q_t}+\epsilon}.
\end{align}
Here $\odot$ means element-wise multiplication, and the division is also
element-wise. RMSProp gives smaller steps to parameters with consistently large
gradients and larger steps to parameters with consistently small gradients. The
important word is ``moving'': $q_t$ is an exponential moving average, so old
gradients are gradually forgotten. Thus RMSProp is adaptive, but it is not a
lifetime-count optimizer.

\paragraph{Adam.}
Adam combines momentum and RMSProp. To avoid confusion with the intrinsic reward
coefficient $\beta$, write Adam's two decay rates as $\rho_1$ and $\rho_2$:
\begin{align}
m_t
&=
\rho_1 m_{t-1}+(1-\rho_1)h_t,\\
q_t
&=
\rho_2 q_{t-1}+(1-\rho_2)(h_t\odot h_t),\\
\widehat{m}_t
&=
\frac{m_t}{1-\rho_1^t},
\qquad
\widehat{q}_t
=
\frac{q_t}{1-\rho_2^t},\\
\phi_{t+1}
&=
\phi_t
-
\eta
\frac{\widehat{m}_t}{\sqrt{\widehat{q}_t}+\epsilon}.
\end{align}
Adam is the standard default optimizer for many neural networks because it is
stable and fast. Adam is a first-order stochastic optimizer that uses
adaptive moment estimates of the gradients \citep{kingma2014adam}. The current RND predictor uses constant-rate Adam with its own
learning rate. But Adam also uses exponential moving averages. Therefore, Adam
does not remember the total number of visits to an input. This means it can make
the RND bonus saturate instead of continuing to decay like $1/\sqrt{n}$.

\paragraph{AdaGrad.}
AdaGrad is different because it uses a cumulative sum of squared gradients:
\begin{align}
c_t
&=
c_{t-1}+h_t\odot h_t,\\
\phi_{t+1}
&=
\phi_t
-
\eta
\frac{h_t}{\sqrt{c_t}+\epsilon}.
\end{align}
Unlike Adam and RMSProp, the accumulator $c_t$ does not forget old gradients. If
the same input, or the same feature direction, appears many times, then the
corresponding entries of $c_t$ grow. The effective step size shrinks roughly like
\[
\frac{1}{\sqrt{c_t}}.
\]
This makes AdaGrad the closest off-the-shelf optimizer to a soft count. It is not
an exact per-state counter, because neural-network parameters are shared across
states, but it is much closer to count accumulation than Adam.

This matches AdaGrad's original motivation: the update adapts to the geometry of
data observed in earlier iterations and gives frequently observed features smaller
effective steps than rare features \citep{duchi2011adaptive}.


\paragraph{A $1/t$ learning-rate schedule.}
Another way to obtain count-like behavior is to use a decreasing learning rate:
\begin{align}
\eta_t
=
\frac{\eta_0}{t+t_0},
\qquad
\phi_{t+1}
=
\phi_t-\eta_t h_t.
\end{align}
This is the classical stochastic-approximation idea: early updates move a lot,
while later updates average more and more slowly. In the idealized case where the
same state is visited $n$ times and the predictor behaves like a local average,
a $1/n$ step size gives an error norm that decays like
\[
\lVert e_n\rVert_2\propto \frac1{\sqrt{n}}.
\]
This matches the count oracle's bonus scale.

\paragraph{Polyak--Ruppert averaging.}
Polyak--Ruppert averaging keeps training with noisy updates, but uses the average
of the iterates for prediction:
\begin{align}
\bar{\phi}_t
=
\frac1t\sum_{k=1}^{t}\phi_k.
\end{align}
The predictor readout becomes $g_{\bar{\phi}_t}$ rather than $g_{\phi_t}$. The
averaging can reduce the variance of the final predictor.

The scope of the $1/\sqrt{n}$ claim must be stated carefully.  The classical
result \citep{polyak1992acceleration} assumes stochastic gradient descent with
decreasing steps $\eta_t\propto t^{-\alpha}$, $\alpha\in(1/2,1)$, on a smooth
problem whose iterates converge to an isolated stationary point with a
positive-definite Hessian, under stationary noise; the averaged iterate is then
asymptotically normal with the optimal covariance, i.e., it attains the
$1/\sqrt{n}$ statistical error rate.  A nonlinear predictor network violates
these assumptions in three ways:
\begin{enumerate}[label=(\roman*),leftmargin=*]
\item \textbf{Non-convexity.}  Convergence to a single good stationary point is
not guaranteed, so the result is at best conditional on the iterates settling
into one basin.
\item \textbf{Degenerate Hessians.}  An overparameterized network has continuous
symmetries (for example, permutations of hidden units), so its minima form flat
manifolds with singular Hessians, and the asymptotic-normality result does not
hold as stated at such minima.
\item \textbf{Parameter averaging is not function averaging.}  For a nonlinear
$g$, $g_{\bar{\phi}_t}(x)\neq\frac1t\sum_{k=1}^{t} g_{\phi_k}(x)$; the parameter
average tracks the averaged predictor only while the iterates stay in a
neighborhood where $g$ is approximately linear in $\phi$.  Iterates drifting
along a symmetry manifold can average to a worse parameter vector, which is one
reason the averaging start $t_{\mathrm{avg}}$ must be swept.
\end{enumerate}
For the linear RND head the predictor objective is least squares---convex and
quadratic---so the classical setting applies exactly and the $1/\sqrt{n}$ rate
is a real prediction, up to the nonstationarity of the RND input stream (which
no optimizer variant escapes).  For the MLP predictor the rate is expected only
insofar as training stays close to initialization, where the network is
approximately linear in its parameters.  Otherwise the honest reading of the
averaged-readout variants is as an empirical variance-reduction and
stabilization test, not as a method with a guaranteed count-scale rate.


\paragraph{Why constant Adam does not give a count bonus.}
Consider one fixed input $x$ that has been included in $n$ predictor updates.
Let
\[
e_n=g_\phi(\tilde{x})-f_\theta(\tilde{x})
\]
be the RND residual at that input after its $n$-th relevant update. A local
linear approximation gives the recursion
\begin{align}
e_{n+1}
\approx
(I-\eta_n K)e_n+\eta_n\xi_n,
\end{align}
where $K$ is the local tangent-kernel curvature and $\xi_n$ represents noise from
other samples in the minibatch.

With a constant effective step size $\eta_n=\eta$, the deterministic part can
decrease quickly, but the noise term keeps injecting error. The residual typically
approaches a nonzero steady-state level:
\[
\mathbb{E}\lVert e_n\rVert_2^2
\longrightarrow
\text{constant depending on }\eta.
\]
So the bonus stops decreasing.

With a count-like step size $\eta_n\propto 1/n$, the recursion behaves like a
running average:
\[
\mathbb{E}\lVert e_n\rVert_2^2
=
O\!\left(\frac1n\right),
\qquad
\lVert e_n\rVert_2
=
O\!\left(\frac1{\sqrt{n}}\right).
\]
Thus the \emph{L2 residual} has the same scaling as the count oracle. If the
bonus is instead the squared loss
\[
d(x)=\frac12\lVert e_n\rVert_2^2,
\]
then the bonus decays like $1/n$, not $1/\sqrt{n}$. Therefore, optimizer choice
and bonus readout must be tested together.

\paragraph{Important distinction: error norm versus squared error.}
The count oracle uses
\[
r^{\mathrm{int}}(n)=\min(1,n^{-1/2}),
\]
which is a standard-deviation-scale quantity. Therefore, the matching learned
quantity is the residual norm
\[
\lVert g_\phi(\tilde{x})-f_\theta(\tilde{x})\rVert_2.
\]
The default RND squared-error bonus
\[
\frac12\lVert g_\phi(\tilde{x})-f_\theta(\tilde{x})\rVert_2^2
\]
is variance-scale. If the norm decays as $1/\sqrt{n}$, the squared error decays
as $1/n$. This is why the next optimizer experiments should include both the
optimizer and the readout type.


\paragraph{Why not optimize the L2 norm directly?}
It is mathematically possible to train the predictor with an L2-norm objective,
for example
\[
L_t^{\mathrm{norm}}(\phi)
=
\frac1b
\sum_{i=1}^{b}
\left\lVert
g_\phi(\tilde{x}_i)-f_\theta(\tilde{x}_i)
\right\rVert_2.
\]
A smoothed version would be
\[
L_t^{\mathrm{smooth\text{-}norm}}(\phi)
=
\frac1b
\sum_{i=1}^{b}
\sqrt{
\left\lVert
g_\phi(\tilde{x}_i)-f_\theta(\tilde{x}_i)
\right\rVert_2^2+\epsilon
}.
\]
However, this is not the usual RND or regression objective. The exact norm is not
differentiable at zero residual. Away from zero, its gradient with respect to the
residual is
\[
\nabla_e \lVert e\rVert_2
=
\frac{e}{\lVert e\rVert_2},
\]
so the gradient magnitude does not shrink in proportion to the residual magnitude.
By contrast, the squared loss has gradient
\[
\nabla_e \frac12\lVert e\rVert_2^2=e,
\]
which is smooth and naturally becomes small as the predictor approaches the
target. This makes squared loss easier to optimize near the solution and gives
the usual least-squares estimator.



The statistical reason is also important. Squared loss corresponds to estimating a
conditional mean and has the standard least-squares/variance interpretation. In a
fixed-feature linear model, minimizing squared error leads to covariance and
confidence formulas involving
\[
(\Phi^\top\Phi+\lambda I)^{-1},
\]
and the uncertainty for a feature vector $\varphi$ is a standard-error quantity
of the form
\[
\sqrt{\varphi^\top(\Phi^\top\Phi+\lambda I)^{-1}\varphi}.
\]
That is exactly the square-root scale used by elliptical bonuses. Directly
optimizing a norm loss is closer to an absolute-error or robust-regression
objective: it can be useful when one wants robustness to outliers, but it is less
aligned with the variance and confidence-bound theory that motivates count-like
RND and elliptical bonuses \citep{huber1964robust,abbasi2011online}.



Therefore the clean separation is:
\[
\boxed{
L_t^{\mathrm{mse}}(\phi)
=
\frac1b\sum_i\frac12\lVert e_\phi(\tilde{x}_i)\rVert_2^2
\quad
\text{for predictor training}
}
\]
and, for the count-scale experiment,
\[
\boxed{
\mathcal{B}_{\mathrm{l2}}(x;\phi)
=
\lVert e_\phi(\tilde{x})\rVert_2
\quad
\text{for the SAC intrinsic reward}
}.
\]
The two functions are different functions of the same residual. They are coupled,
but they play different roles.

\paragraph{Optimizer conclusion.}
Adam is a good default for making neural-network losses decrease, but it is not
designed to represent visit counts. For RND as an exploration bonus, the next
experiments should test count-aware alternatives: AdaGrad, SGD with a $1/t$
schedule, and Polyak--Ruppert averaging. These should be paired with an L2
residual readout when the goal is to match the count oracle's $1/\sqrt{n}$ scale.


The final design rule is therefore:
\[
\begin{aligned}
&\boxed{\text{train the RND predictor with MSE}}
\quad\text{and}\\[6pt]
&\boxed{\;\begin{array}{c}
\text{use the L2 residual as the bonus only}\\
\text{when the desired bonus scale is }1/\sqrt{n}
\end{array}\;}.
\end{aligned}
\]
Directly optimizing the L2 norm is possible, but it changes the regression
problem: it gives a nonsmooth or smoothed-norm objective rather than the standard
least-squares objective, weakens the variance/confidence interpretation, and is
less natural for the elliptical-bonus connection. In the proposed experiments,
the loss function and the bonus function should therefore be two functions of the
same residual: squared residual for training, residual norm for count-scale
reward readout.


\subsection{Optimizers used in practice}
\label{sec:opt-used}

Every RND implementation we examined trains the predictor with Adam
\citep{kingma2014adam}; Table~\ref{tab:opt-used} lists the settings. They agree
on betas $(0.9,0.999)$ and zero weight decay and differ only in the columns
shown. The original RND line of work---the paper \citep{burda2019exploration},
OpenAI's release, the most-used PyTorch port, and CleanRL---folds the predictor
into the \emph{same} Adam as the PPO policy at learning rate $10^{-4}$. This
project and RLeXplore instead give the predictor its \emph{own} Adam at
$10^{-3}$, because SAC \citep{haarnoja2018soft} is off-policy and the predictor
is fit on replay minibatches rather than on-policy rollouts. The fact that
matters for Section~\ref{sec:opt-defs}: none of them anneal the predictor's
learning rate on a $1/n$ schedule and none use a cumulative (AdaGrad-style)
step---the learning rate is constant throughout (CleanRL's optional
linear-to-zero anneal is the only schedule, and it decays faster than $1/n$, not
slower).

\section{Logging}
\label{sec:logging}
%==============================================================================

Every training run records its metrics to \textbf{its own JSON file} under
\texttt{train\_\allowbreak runs/<run>/\allowbreak data/\allowbreak <sweep\_\allowbreak id>/local/}. Because each
run owns a distinct file, the $512$ concurrent work-queue workers never write the same path, so there is
\textbf{no file-lock contention}. File access is slow, so logging is \textbf{group-style and written once}: every metric is
accumulated in memory by its callback at the evaluation cadence (\texttt{eval\_freq}) during training, and
the whole record --- all groups --- is flushed in a \textbf{single file write at the end of the run}. No
metric is written incrementally per step.

\paragraph{Run id (filename convention).}
The local work queue replaces the W\&B sweep as the config source. Each run carries an integer
\texttt{run\_id} $\in\{0,\dots,\texttt{run\_total}-1\}$ giving its position in the sweep, and
\texttt{run\_total} (here $600$); together \texttt{run\_id}/\texttt{run\_total} is the run's identity and
names its JSON file (e.g.\ \texttt{042\_of\_600.json}). The sweep iterates the \textbf{seed outermost}:
seed $s$ owns a contiguous id block of one id per algorithm in fixed order (\texttt{gt\_\allowbreak
position\_\allowbreak velocity}, \texttt{rnd\_\allowbreak elliptical}, \texttt{rnd\_\allowbreak state}), so
seed $0$ takes ids $0$--$2$, seed $1$ takes $3$--$5$, and an earlier seed always has smaller ids than a
later one.

\paragraph{Sweep id (many sweeps per run folder).}
A run folder may hold \textbf{many sweeps} --- reruns of the same configuration (which \emph{extend} coverage)
or different configurations. Each sweep launch is given a \texttt{sweep\_\allowbreak id} of the form
\texttt{<date>\_\allowbreak <tag>} (the tag names the configuration version), and everything it produces is
scoped under that id: its queue under \texttt{queue/<sweep\_\allowbreak id>/}, its per-run JSONs under
\texttt{data/<sweep\_\allowbreak id>/local/}, and its submitted job ids in
\texttt{submitted\_\allowbreak jobids\_\allowbreak <sweep\_\allowbreak id>.txt}. A manifest
(\texttt{data/SWEEPS.md}) catalogues every sweep. The analysis loads a sweep by id (pooling reruns that share a
configuration tag), so superseded or invalid data in the same folder never contaminates a new analysis --- and
the run id above is sweep-local, with the pair \texttt{(sweep\_\allowbreak id, run\_\allowbreak id)} unique
within the folder.

\paragraph{What is logged.}
Table~\ref{tab:logging} lists every logged quantity by group. A \emph{group} is the set of metrics written
together as one JSON list; grouping is what minimizes the number of slow file writes. An
\emph{intermediate} metric is captured at every \texttt{eval\_freq} step (one snapshot appended to its
group's list) and persisted at the final write; a \emph{final} field is written once at run end.

\paragraph{Aggregation.}
Finished runs are grouped by $(\texttt{algorithm}, \beta)$. The primary metric is
the final eval extrinsic reward $r^{\mathrm{ext}}$ (the $10^{4}$-episode evaluation
at $2\times10^{6}$ steps). For a group of $n$ seeds with per-seed final rewards
$R_i$, we report the seed mean and its standard error,
\begin{align}
\bar{R} = \frac{1}{n}\sum_{i=1}^{n} R_i,
\qquad
\mathrm{SE} = \frac{s}{\sqrt{n}},
\qquad
s = \sqrt{\frac{1}{n-1}\sum_{i=1}^{n}\big(R_i-\bar{R}\big)^2}.
\end{align}
For each algorithm we keep the single coefficient
$\beta^\star = \argmax_{\beta}\,\bar{R}(\texttt{algorithm},\beta)$ with the largest
mean reward---the best-tuned coefficient---and report that group.

\subsection{Train run 3: elliptical bonuses and RND optimizers (design and runs)}
\label{sec:train-run-3-proposed}

\paragraph{Purpose.}
Train run~3 tests two count-matching hypotheses.  The first hypothesis is that the
elliptical bonus works because
\[
\sqrt{\varphi^\top\Lambda^{-1}\varphi}
\]
is a continuous analogue of the count bonus \(1/\sqrt{n}\).  The second hypothesis
is that ordinary RND is limited because constant-rate Adam does not accumulate a
visit count.  Thus the run is divided into two parts:
\[
\begin{aligned}
&\boxed{\text{Part I: how should the elliptical matrix }\Lambda\text{ remember data?}}\\[6pt]
&\boxed{\text{Part II: how should the RND predictor be optimized?}}
\end{aligned}
\]

\paragraph{Priority convention.}
All sweep tables use the same priority notation:
\[
\textbf{must try}
\quad>\quad
\text{try}
\quad>\quad
\textcolor{gray}{\text{less interesting}}.
\]
The full Cartesian product should \emph{not} be run.  A \textbf{must try} row
belongs in the first-stage sweep.  A plain row is a second-stage refinement after
the best family is found.  A \textcolor{gray}{gray} row is a diagnostic or
stability check, not part of the main grid.

\paragraph{Run numbering.}
This subsection holds the two \emph{design} parts first (Part~I elliptical, Part~II RND optimizers,
then the final-run design), followed by the \emph{executed} runs, appended in launch order and numbered
\(3.\langle\text{family}\rangle.\langle\text{index}\rangle\): family~1 = elliptical bonuses (Part~I),
family~2 = RND optimizer variants (Part~II). Executed so far: Train run~3.1.1 (four-method elliptical/RND
sweep) and Train run~3.1.2 (run-2 replication and factor ablation).

\paragraph{Raw features and normalized features.}
Let
\[
z_\varphi(x)\in\mathbb{R}^{d_\varphi}
\]
be the raw output of the frozen random encoder.  The matrix \(\Lambda\) should not
be built from \(z_\varphi(x)\) directly, because the raw feature norm is an
uncontrolled random-network artifact.  If one input has a larger raw norm than
another, then
\[
z_\varphi(x)^\top\Lambda^{-1}z_\varphi(x)
\]
can be large because of feature amplitude, not because of genuine novelty.

Therefore the covariance matrix will be built from a normalized feature
\[
\varphi(x)
=
\mathcal{N}_{\nu_\varphi}\!\left(z_\varphi(x)\right),
\]
where \(\nu_\varphi\) is the feature-normalization mode.  The default mode is
unit-norm normalization:
\begin{align}
\mathcal{N}_{\mathrm{unit}}\!\left(z\right)
&=
\frac{z}{\lVert z\rVert_2+\epsilon_\varphi},
\qquad
\epsilon_\varphi=10^{-8}.
\end{align}
Then
\[
\lVert \varphi(x)\rVert_2\le 1.
\]
This makes the scale of \(\lambda\) interpretable.  In particular, if
\(\Lambda_0=\lambda I_{d_\varphi}\), then
\[
\varphi(x)^\top\Lambda_0^{-1}\varphi(x)
=
\frac{\lVert\varphi(x)\rVert_2^2}{\lambda}
\le
\frac1\lambda,
\qquad
\mathcal{E}_0(x)\le \frac1{\sqrt{\lambda}}.
\]
Thus, with unit-norm features, \(\lambda=1\) gives an initial maximum bonus of
approximately \(1\), matching the scale of the clipped count oracle.

A second normalization mode is coordinate RMS followed by unit normalization:
\begin{align}
u_t(z)
&=
\frac{z-\mu_t^\varphi}{\sigma_t^\varphi+\epsilon_\varphi},\\
\mathcal{N}_{\mathrm{rms\text{-}unit}}(z)
&=
\frac{u_t(z)}{\lVert u_t(z)\rVert_2+\epsilon_\varphi}.
\end{align}
Here \(\mu_t^\varphi\) and \((\sigma_t^\varphi)^2\) are running feature statistics.
They are updated from the replay minibatch \(B_t\) sampled from the buffer---the
same batch used to compute the bonus and update the covariance, since the bonus is
recomputed and these statistics refreshed every time a batch is sampled from the
replay buffer.  This mode removes coordinate-scale imbalance before the final unit
normalization.  It is more engineered than pure unit-norm normalization, so it is a
second-stage option.

The unnormalized mode
\[
\mathcal{N}_{\mathrm{none}}(z)=z
\]
is kept only as a diagnostic.  It makes \(\lambda\) difficult to interpret and can
produce large intrinsic-reward spikes.

\paragraph{Common elliptical bonus.}
For any positive-definite matrix
\[
\Lambda_t\in\mathbb{R}^{d_\varphi\times d_\varphi},
\qquad
\Lambda_t=\Lambda_t^\top,
\qquad
v^\top\Lambda_t v>0
\quad\text{for all }v\ne0,
\]
define the exact quadratic form
\begin{align}
q_t^{\mathrm{exact}}(x;\Lambda_t)
&=
\varphi(x)^\top\Lambda_t^{-1}\varphi(x).
\end{align}
The exact elliptical bonus is
\begin{align}
\mathcal{E}_t^{\mathrm{exact}}(x;\Lambda_t)
&=
\sqrt{q_t^{\mathrm{exact}}(x;\Lambda_t)}.
\end{align}
The square root is essential.  If \(\varphi(x)=e_j\) is a one-hot cell feature and
\[
\Lambda_t
=
\lambda I_{d_\varphi}
+
\sum_{\tau\le t}
\varphi(x_\tau)\varphi(x_\tau)^\top,
\]
then
\[
\Lambda_t
=
\operatorname{diag}(\lambda+n_1,\ldots,\lambda+n_{d_\varphi}),
\]
and for a point in cell \(j\),
\begin{align}
\mathcal{E}_t^{\mathrm{exact}}(x;\Lambda_t)
&=
\sqrt{
e_j^\top\Lambda_t^{-1}e_j
}
=
\frac{1}{\sqrt{\lambda+n_j}}.
\end{align}
Thus the elliptical bonus is a soft feature-space version of the count bonus
\(1/\sqrt{n}\).  This is the same square-root quadratic form used by
least-squares confidence ellipsoids in linear bandits \citep{abbasi2011online}.

\paragraph{Stable quadratic-form computation.}
The notation \(\Lambda_t^{-1}\varphi(x)\) is mathematical shorthand.  It should
not be implemented by explicitly computing \(\Lambda_t^{-1}\).  The stable
computation is a linear solve.

First define the symmetric part of a matrix:
\[
\operatorname{sym}(M)
=
\frac12(M+M^\top).
\]
Before factorization, form
\begin{align}
A_t
&=
\operatorname{sym}(\Lambda_t)
+
\epsilon_{\mathrm{chol}} I_{d_\varphi}.
\end{align}
The constant \(\epsilon_{\mathrm{chol}}\ge0\) is \emph{Cholesky jitter}.  It is a
purely numerical diagonal correction used to make the factorization robust to
floating-point round-off.  It is not the ridge regularizer \(\lambda\).  The
default implementation should try
\[
\epsilon_{\mathrm{chol}}=0
\]
first and retry with
\[
\epsilon_{\mathrm{chol}}=10^{-8}
\]
only if Cholesky factorization fails.  Larger jitter values should be treated as
debugging evidence that the covariance construction is unstable, not as a normal
sweep dimension.

Since \(A_t\) is positive definite, compute its Cholesky factorization
\begin{align}
A_t
&=
L_tL_t^\top,
\end{align}
where \(L_t\) is lower triangular with positive diagonal entries.  Then solve
\begin{align}
L_t y_t(x)
&=
\varphi(x).
\end{align}
The computed quadratic form is
\begin{align}
\widehat q_t(x;\Lambda_t)
&=
\lVert y_t(x)\rVert_2^2.
\end{align}
Indeed,
\begin{align}
\lVert L_t^{-1}\varphi(x)\rVert_2^2
&=
\varphi(x)^\top L_t^{-\top}L_t^{-1}\varphi(x)\\
&=
\varphi(x)^\top (L_tL_t^\top)^{-1}\varphi(x)\\
&=
\varphi(x)^\top A_t^{-1}\varphi(x).
\end{align}
Thus the implementation computes the quadratic form by factorization and triangular
solve, not by an explicit inverse.

Finally define the implemented bonus as
\begin{align}
\mathcal{E}_t(x;\Lambda_t)
&=
\sqrt{
\max\!\left(\widehat q_t(x;\Lambda_t),\,\epsilon_q\right)
}.
\end{align}
In exact arithmetic, \(q_t^{\mathrm{exact}}\ge0\).  The maximum is only a guard
against round-off.  The default is
\[
\epsilon_q=0.
\]
If a strictly positive value is needed for logging or normalization, use
\(\epsilon_q=10^{-12}\); do not sweep it.

\paragraph{Batch evaluation.}
For a batch \(C=\{x_i\}_{i=1}^c\), stack features as rows:
\[
\Phi_C
=
\begin{bmatrix}
\varphi(x_1)^\top\\
\vdots\\
\varphi(x_c)^\top
\end{bmatrix}
\in\mathbb{R}^{c\times d_\varphi}.
\]
After computing \(A_t=L_tL_t^\top\), solve one triangular matrix equation
\[
L_tY^\top=\Phi_C^\top,
\qquad
Y\in\mathbb{R}^{c\times d_\varphi}.
\]
Then
\[
\widehat q_t(x_i;\Lambda_t)
=
\sum_{k=1}^{d_\varphi}Y_{ik}^2.
\]
This vectorized solve is the default implementation for replay minibatches.

\paragraph{Reduced numerical knobs.}
The ridge \(\lambda\) and the jitter \(\epsilon_{\mathrm{chol}}\) have different
roles:
\[
\boxed{
\lambda I
\text{ is the statistical prior in }\Lambda
}
\qquad
\boxed{
\epsilon_{\mathrm{chol}}I
\text{ is a last-mile numerical correction before factorization.}
}
\]
Because unit-norm features make \(\lambda\) interpretable, we do not need a large
ridge sweep.  The cumulative methods should use \(\lambda=1\) as the first-stage
value.  This is the count-prior choice: one unobserved unit-norm direction has
bonus at most \(1\).  The batch method may use the smaller value
\(\lambda=10^{-2}\), because its covariance is averaged by \(1/b\) and otherwise
the bonus can become too flat.  Values near \(10^{-6}\) are deliberately removed
from the main sweep; after unit normalization they mostly test numerical
aggressiveness, not a distinct statistical idea.

\paragraph{Method E1: batch elliptical.}
Let
\[
B_t=\{x_i\}_{i=1}^{b}
\]
be the replay minibatch at update \(t\).  The batch covariance is
\begin{align}
\Lambda_t^{\mathrm{batch}}
&=
\lambda I_{d_\varphi}
+
\frac1b
\sum_{x_i\in B_t}
\varphi(x_i)\varphi(x_i)^\top\\
&=
\lambda I_{d_\varphi}
+
\frac1b
\Phi_{B_t}^\top\Phi_{B_t}.
\end{align}
The factor \(1/b\) makes this matrix approximately independent of minibatch size.
This method measures novelty relative to the current replay minibatch only.

\paragraph{Instabilities and remedies for E1.}
The batch matrix can be low-rank because a minibatch need not span
\(\mathbb{R}^{d_\varphi}\).  The ridge \(\lambda I\) makes the matrix positive
definite.  Small \(\lambda\) gives sharper novelty but can produce large rewards
in feature directions absent from the batch.  With unit-norm features, the first
batch value to test is
\[
\lambda=10^{-2}.
\]
The value \(\lambda=1\) is a conservative check.  The old very small ridge values
are not first-stage candidates.

The conceptual limitation is not numerical: E1 does not remember lifetime visits.
A state can receive a large bonus simply because its feature direction is absent
from the current minibatch, even if the state has been visited many times before.
Therefore E1 is a baseline, not the main count-like method.

\paragraph{Method E2: global elliptical.}
Let \(B_t\) be the replay minibatch sampled from the buffer at update \(t\).  The
global covariance accumulates the feature outer-products of every sampled batch:
\begin{align}
\Lambda_t^{\mathrm{global}}
&=
\Lambda_{t-1}^{\mathrm{global}}
+
\sum_{x_i\in B_t}
\varphi(x_i)\varphi(x_i)^\top,
\qquad
\Lambda_0^{\mathrm{global}}=\lambda I_{d_\varphi}.
\end{align}
Equivalently,
\[
\Lambda_t^{\mathrm{global}}
=
\lambda I_{d_\varphi}
+
\sum_{\tau\le t}
\Phi_{B_\tau}^\top\Phi_{B_\tau}.
\]
Because the covariance is updated on each batch sampled from the replay buffer, it
accumulates feature coverage over all sampled transitions, giving the lifetime
(cumulative) feature-space memory that distinguishes E2 from the per-batch E1.  A
transition contributes once for each time it is sampled, so the matrix weights
transitions by how often they are replayed.

\paragraph{Instabilities and remedies for E2.}
The naive cumulative sum would require revisiting all past data.  The stable
implementation is the rank-batch update
\[
\Lambda_t
=
\Lambda_{t-1}
+
\Phi_{B_t}^\top\Phi_{B_t},
\]
which costs \(O(|B_t|d_\varphi^2)\) time and \(O(d_\varphi^2)\) memory.  The
Cholesky factorization costs \(O(d_\varphi^3)\) and is recomputed on every
covariance update; there is no refresh cadence to tune.  Nothing is saved by
amortizing it: the batch method (E1) already pays the same \(O(d_\varphi^3)\)
factorization on every sampled minibatch, so refactoring the global covariance
once per update --- the same cost --- has nothing to amortize against.

The ridge default is \(\lambda=10^{-2}\), the same value as the batch method.  The
count-prior value \(\lambda=1\) (which, with unit-norm features, acts like one
prior visit in every feature direction and bounds the initial bonus by \(1\)) is
kept only as an optional sweep point, not the default.

\paragraph{Method E3: discounted global elliptical.}
The discounted method gives old visits less weight:
\begin{align}
\Lambda_t^{\mathrm{disc}}
&=
\rho_\Lambda\Lambda_{t-1}^{\mathrm{disc}}
+
\sum_{x_i\in B_t}
\varphi(x_i)\varphi(x_i)^\top
+
(1-\rho_\Lambda)\lambda I_{d_\varphi},
\qquad
0<\rho_\Lambda\le1.
\end{align}
The term \((1-\rho_\Lambda)\lambda I_{d_\varphi}\) is necessary.  Without it, the
baseline ridge would decay away.  With it, the matrix can be written as
\[
\Lambda_t^{\mathrm{disc}}
=
\lambda I_{d_\varphi}
+
\sum_{\tau\le t}
\rho_\Lambda^{t-\tau}
\varphi(x_\tau)\varphi(x_\tau)^\top,
\]
up to the batching convention.  The effective memory length is approximately
\[
H_\Lambda
\approx
\frac{1}{1-\rho_\Lambda}.
\]

\paragraph{Instabilities and remedies for E3.}
If \(\rho_\Lambda\) is too small, the method forgets useful lifetime coverage and
becomes close to a recency bonus.  If \(\rho_\Lambda\) is too close to \(1\), it is
nearly identical to the global method.  Therefore this is not a first-stage
method.  It should be tested only after E2 shows whether lifetime memory is
helpful.

\paragraph{Method E4: mixed batch--global elliptical.}
The mixed method combines local minibatch novelty and lifetime novelty.  Let
\[
\mathcal{E}_t^{\mathrm{batch}}(x)
=
\mathcal{E}_t(x;\Lambda_t^{\mathrm{batch}}),
\qquad
\mathcal{E}_t^{\mathrm{global}}(x)
=
\mathcal{E}_t(x;\Lambda_t^{\mathrm{global}}).
\]
Because the two matrices have different scales, raw bonuses should not be mixed
directly.  Instead maintain running second moments
\begin{align}
M_t^{v}
&=
\left(1-\frac1{\tau_{\mathrm{rms}}}\right)M_{t-1}^{v}
+
\frac1{\tau_{\mathrm{rms}}}
\cdot
\frac1{|C_t|}
\sum_{x\in C_t}
\left(\mathcal{E}_t^{v}(x)\right)^2,
\qquad
v\in\{\mathrm{batch},\mathrm{global}\},
\end{align}
where \(C_t\) is the bonus-evaluation batch.  Define
\begin{align}
\widehat{\mathcal{E}}_t^{v}(x)
&=
\frac{\mathcal{E}_t^{v}(x)}
{\sqrt{M_t^v}+\epsilon_{\mathrm{rms}}},
\qquad
v\in\{\mathrm{batch},\mathrm{global}\}.
\end{align}
The mixed bonus is
\begin{align}
\mathcal{E}_t^{\mathrm{mix}}(x)
&=
(1-\omega)\widehat{\mathcal{E}}_t^{\mathrm{batch}}(x)
+
\omega\widehat{\mathcal{E}}_t^{\mathrm{global}}(x),
\qquad
0\le\omega\le1.
\end{align}

\paragraph{Instabilities and remedies for E4.}
The main instability is scale mismatch.  The batch covariance is an averaged
minibatch matrix, while the global covariance is cumulative.  The remedy is to
mix RMS-normalized bonuses, not raw quadratic forms.  The second instability is
early-time division by a small RMS; use
\[
\epsilon_{\mathrm{rms}}=10^{-8}
\]
and initialize \(M_0^v=1\), so the first updates do not create artificial reward
spikes.

The endpoints \(\omega=0\) and \(\omega=1\) duplicate the pure batch and pure
global methods.  They are not part of the main mixed sweep.

\subsubsection{Part II (design): RND optimizer variants}
\label{sec:train-run-3-rnd-optimizers}

\paragraph{Common RND object.}
For an RND input \(x\), define the residual
\[
e_\phi(x)
=
g_\phi(\tilde{x})-f_\theta(\tilde{x}),
\]
where \(f_\theta\) is frozen and \(g_\phi\) is trained.  The predictor objective
is always the mean squared error
\begin{align}
L_t^{\mathrm{mse}}(\phi)
=
\frac1b\sum_{i=1}^{b}
\frac12\lVert e_\phi(x_i)\rVert_2^2.
\end{align}
Only the optimizer and the bonus readout are changed.  The two readouts are
\begin{align}
\mathcal{B}_{\mathrm{mse}}(x;\phi)
&=
\frac12\lVert e_\phi(x)\rVert_2^2,\\
\mathcal{B}_{\mathrm{l2}}(x;\phi)
&=
\lVert e_\phi(x)\rVert_2 .
\end{align}
The MSE readout is the canonical RND bonus \citep{burda2019exploration}.  The L2
readout is the count-scale readout: if
\(\mathbb{E}\lVert e_n\rVert_2^2=O(1/n)\), then
\(\lVert e_n\rVert_2=O(1/\sqrt{n})\).  A constant rescaling of the L2 readout
(for example a \(1/\sqrt{m}\) output-dimension normalization) is absorbed into
the swept coefficient \(\beta\) and leaves every distance-to-ground-truth metric
unchanged (all of them fit or remove a scalar factor), so no normalized variant
is kept.

\paragraph{Common optimizer difficulties.}
The predictor loss and the intrinsic reward must not be confused.  SAC receives
\[
r_t^{\mathrm{train}}
=
r_t^{\mathrm{ext}}
+
\beta\,\mathcal{B}(x_t;\phi_t),
\]
but the RND optimizer differentiates \(L_t^{\mathrm{mse}}\), not
\(\mathcal{B}\) directly.  We do \emph{not} switch the predictor objective to
\(\lVert e\rVert_2\), because the exact norm is nonsmooth at zero and its
gradient \(e/\lVert e\rVert_2\) does not vanish with the residual.  The clean
design is:
\[
\boxed{\text{train with squared loss}}
\qquad
\boxed{\text{read out either squared error or L2 residual as reward}} .
\]

%------------------------------------------------------------------------------
\paragraph{Method O1: Adam.}
Adam keeps exponential moving averages of the gradient and squared gradient:
\begin{align}
m_t
&=
\rho_1 m_{t-1}+(1-\rho_1)h_t,\\
q_t
&=
\rho_2 q_{t-1}+(1-\rho_2)(h_t\odot h_t),\\
\widehat{m}_t
&=
\frac{m_t}{1-\rho_1^t},
\qquad
\widehat{q}_t
=
\frac{q_t}{1-\rho_2^t},\\
\phi_{t+1}
&=
\phi_t
-
\eta_{\mathrm{Adam}}
\frac{\widehat{m}_t}{\sqrt{\widehat{q}_t}+\epsilon_{\mathrm{Adam}}},
\qquad
h_t=\nabla_\phi L_t^{\mathrm{mse}}(\phi_t).
\end{align}
This is the practice baseline for RND and neural-network regression
\citep{kingma2014adam,burda2019exploration}.

\paragraph{Instabilities and remedies for O1.}
Adam is stable for reducing a neural loss, but it is not a lifetime-count
optimizer.  Both \(m_t\) and \(q_t\) are exponentially weighted, so old visits are
forgotten.  With a constant effective step size, the residual can approach a
nonzero steady state rather than decaying like \(1/\sqrt{n}\).  Adam must
therefore be run in two forms: the canonical Adam-MSE baseline and the Adam-L2
readout ablation.  The first tests standard RND; the second tests whether the
readout scale alone explains the gap.

\paragraph{Method O2: AdaGrad.}
AdaGrad uses a cumulative sum of squared gradients:
\begin{align}
c_t
&=
c_{t-1}+h_t\odot h_t,\\
\phi_{t+1}
&=
\phi_t
-
\eta_{\mathrm{Ada}}
\frac{h_t}{\sqrt{c_t}+\epsilon_{\mathrm{Ada}}}.
\end{align}
Unlike Adam, \(c_t\) is not an exponential moving average.  It remembers all past
squared gradients.  Thus repeated feature directions get smaller effective steps,
which is the closest off-the-shelf optimizer analogue of a soft count
\citep{duchi2011adaptive}.

\paragraph{Instabilities and remedies for O2.}
AdaGrad can fail in two opposite ways.  If \(c_0=0\) and the early gradients are
small, the first steps can be too large because the denominator is tiny.  If
\(c_t\) grows too quickly, the predictor can stall because the effective learning
rate becomes too small.  The remedies are an initial accumulator \(c_0\), a
larger base learning rate than Adam, and gradient clipping.  In this run these
knobs are not tuned: AdaGrad uses the PyTorch defaults
(\(\eta_{\mathrm{Ada}}=10^{-2}\), \(c_0=0\), \(\epsilon_{\mathrm{Ada}}=10^{-10}\)),
which already give a base learning rate \(10\times\) Adam's.  The main readout
should be \(\mathcal{B}_{\mathrm{l2}}\), because AdaGrad is being tested as a
count-memory optimizer.

\paragraph{Method O3: SGD with a \(1/t\) schedule.}
The decreasing-step method uses
\begin{align}
\eta_t
&=
\max\!\left(
\eta_{\min},
\frac{\eta_0}{1+t/t_0}
\right),\\
\phi_{t+1}
&=
\phi_t-\eta_t h_t.
\end{align}
Here \(t\) is the predictor-update index.  Concretely, \(t\) starts at \(0\) and
increases by \(1\) on every SAC gradient step, because each replay-minibatch
draw triggers exactly one predictor gradient step on that same minibatch
(\texttt{VectorIntrinsicReplayBuffer.sample()}); with this project's cadence
(\texttt{train\_freq=1}, \texttt{gradient\_steps=1}) that is one predictor
update per environment step, so a \(10^6\)-step run ends near \(t=10^6\).  Note
\(t\) is a single global counter, not a per-state visit count: the step size
decays with total training progress even for states visited for the first time.
The offset \(t_0\) prevents the step
size from becoming small before the predictor has learned the target.

\paragraph{Comparison with the linear-anneal baseline.}
The only decaying baseline in Table~\ref{tab:opt-used} is CleanRL's linear
anneal
\begin{align}
\eta_u
=
\eta_0\left(1-\frac{u-1}{U}\right)
=
\eta_0\left(1-\frac{t}{U}\right),
\end{align}
where \(u\) is the update index, \(U\) is the total number of updates,
\(\eta_0\) is the initial learning rate, and \(t=u-1\) is the same zero-based
update counter as above.  Written side by side, the two schedules differ only
in how the time ratio enters:
\(\eta_0\left(1-t/U\right)\) versus
\(\eta_0/\!\left(1+t/t_0\right)\).  The linear anneal falls at a constant
rate set by the training budget and reaches exactly \(0\) at the end of the
run.  The shifted \(1/t\) schedule is budget-independent and never reaches
zero: it halves at \(t=t_0\) and then follows the long \(\eta_0 t_0/t\) tail
that produces the running-average (count-style) behavior.  Late in training the
two are opposite: the linear anneal freezes the predictor entirely, while the
\(1/t\) schedule keeps updating it with ever-smaller steps.

\paragraph{Instabilities and remedies for O3.}
A pure \(1/t\) schedule can freeze too early.  Conversely, a large \(\eta_0\) can
make the reward oscillate because the predictor changes the intrinsic reward
rapidly.  The mathematical remedy is the shifted schedule
\(\eta_0/(1+t/t_0)\), plus an optional floor \(\eta_{\min}\).  The engineering
remedy is to log the predictor loss and intrinsic-reward RMS separately from the
SAC return.  This method is theoretically attractive because stochastic
approximation with a decreasing step can behave like a running average, but it is
less adaptive than Adam or AdaGrad.

\paragraph{Method O4: Adam with Polyak--Ruppert readout.}
This method trains the predictor parameters \(\phi_t\) with Adam, but reads the
bonus from an averaged predictor.  After a burn-in time \(t_{\mathrm{avg}}\),
define
\begin{align}
\bar{\phi}_t
&=
\bar{\phi}_{t-1}
+
\alpha_t(\phi_t-\bar{\phi}_{t-1}),\\
\alpha_t
&=
\frac{1}{t-t_{\mathrm{avg}}+1},
\qquad
t\ge t_{\mathrm{avg}}.
\end{align}
The intrinsic bonus is
\[
\mathcal{B}(x;\bar{\phi}_t)
\]
rather than \(\mathcal{B}(x;\phi_t)\).

\paragraph{Instabilities and remedies for O4.}
Averaging from initialization can preserve a poor early predictor.  Therefore
\(t_{\mathrm{avg}}\) should be swept.  The method also requires one extra copy of
the predictor parameters, but this is only \(O(|\phi|)\) memory and no extra
backward pass.  Adam-PR is not as theoretically clean as SGD-\(1/t\)-PR because
Adam is still a constant-gain optimizer underneath; nevertheless it is cheap and
tests whether readout averaging alone stabilizes the RND bonus.

\paragraph{Method O5: SGD-\(1/t\) with Polyak--Ruppert readout.}
This method combines the decreasing-step update with averaged readout:
\begin{align}
\phi_{t+1}
&=
\phi_t-\eta_t h_t,
\qquad
\eta_t=\max\!\left(\eta_{\min},\frac{\eta_0}{1+t/t_0}\right),\\
\bar{\phi}_t
&=
\bar{\phi}_{t-1}
+
\frac{1}{t-t_{\mathrm{avg}}+1}
(\phi_t-\bar{\phi}_{t-1}).
\end{align}
The bonus is \(\mathcal{B}_{\mathrm{l2}}(x;\bar{\phi}_t)\).

\paragraph{Instabilities and remedies for O5.}
This is the most statistically principled optimizer family---with the caveat
from Section~\ref{sec:opt-defs} that the \(1/\sqrt{n}\) averaging guarantee is a
theorem only for the linear head, not for the nonlinear MLP predictor---but it
has the most
knobs: \(\eta_0\), \(t_0\), \(\eta_{\min}\), and \(t_{\mathrm{avg}}\).  To avoid
a large grid, use the best \((\eta_0,t_0)\) from Method O3, then sweep only
\(t_{\mathrm{avg}}\).  The averaged predictor should be updated without gradient
tracking and should be used only for reward readout, not for further training.

The primary elliptical comparison is
\[
\text{batch memory}
\quad\text{versus}\quad
\text{global lifetime memory}
\quad\text{versus}\quad

\]
The primary optimizer comparison is
\[
\begin{aligned}
&\text{constant-rate Adam}\quad\text{versus}\\[2pt]
&\text{cumulative AdaGrad}\quad\text{versus}\\[2pt]
&\text{decreasing-step SGD}\quad\text{versus}\\[2pt]

\end{aligned}
\]
The final confirmation should rerun the best elliptical method and the best RND
optimizer method with the same seed budget and evaluation protocol used for
Train run~2.

(\(n{=}39\)), so the gap is not explained by the sweep picking different hyperparameters. The two are
\emph{not} the same algorithm: between run~2 and run~3.1.1 the elliptical implementation changed in three
ways. (i)~\emph{Feature normalization}: run~2 used raw frozen-encoder features (unbounded, state-dependent
norm); run~3.1.1 unit-normalizes every feature (\(\lVert\varphi\rVert_2\le1\)).
(ii)~\emph{Ridge}: run~2 hard-coded \(\lambda{=}10^{-6}\); the run-3.1.1 grid contains only
\(\{1, 10^{-2}\}\). (iii)~\emph{Bonus clip}: run~2 had no clip; run~3.1.1 clips at \(5\). Together these
compress the bonus's dynamic range by orders of magnitude: with unit-norm features the bonus is bounded by
\(\sqrt{\varphi^\top\Lambda^{-1}\varphi} \le \sqrt{1/\lambda}\) --- at most \(10\) for
\(\lambda{=}10^{-2}\), further truncated to \(5\) by the clip, and at most \(1\) for \(\lambda{=}1\) ---
whereas run-2's raw features with \(\lambda{=}10^{-6}\) and no clip produced novelty bonuses up to
\(\sim 10^{3}\,\lVert z_\varphi\rVert\) with a strong novel-vs-visited contrast that the coefficient
\(\beta\) alone cannot restore (rescaling shifts the mean level but not the compressed contrast, and the
clip flattens exactly the most-novel states). The metric change (run-2 eval vs.\ run-3.1.1
training-episode reward) contributes at most a small part of the gap: the run-2 \texttt{rnd\_\allowbreak
state} eval reference (\(35.79\)) and the run-3.1.1 \texttt{rnd\_\allowbreak next\_\allowbreak state}
training curve (\(36.95\)) nearly coincide in Figure~\ref{fig:trainrun3-1-1-curve}. Isolating which of the
three changes matters --- e.g.\ re-running the batch elliptical with \(\lambda{=}10^{-6}\), a disabled
clip, or raw features under the current code --- is left to a follow-up run (Train run~3.1.2, next).

\paragraph{The research questions, opened.}
This run answers four questions.
\begin{enumerate}[label=\textbf{RQ\arabic*},leftmargin=*]
\item \textbf{(replication)} If run-2's exact elliptical configuration is rerun under today's code, does
it recover run-2's \(44.49 \pm 3.32\)? If yes, the run-3.1.1 collapse is fully explained by the three
changed knobs; if no, something else (environment, trainer, the float64 solver) changed too.
\item \textbf{(attribution)} Which of the three knobs --- feature normalization (raw \(\to\) unit), ridge
(\(10^{-6} \to \{10^{-2},1\}\)), bonus clip (none \(\to 5\)) --- carries how much of the damage? The full
factorial measures each knob alone and in every combination.
\item \textbf{(mechanism)} WHY does the winning configuration win? Two candidate explanations are spelled
out below (Hypotheses 1 and 2), and each run logs the measured quantities that tell them apart.
\item \textbf{(infrastructure)} At equal silicon, does the 8-tasks\(\times\)2-CPU or the
16\(\times\)1 Slurm job shape train more steps per CPU-hour? (Measured alongside; summary at the end of
this section, details in the run folder's \texttt{analysis.md}.)
\end{enumerate}

\paragraph{The objects in the bonus, defined.}
Every symbol used below is defined here, in the order it enters the bonus.
\begin{itemize}
\item \textbf{The encoder} \(z(\cdot)\): a frozen, randomly initialized two-layer network mapping the
state--action pair \(x = (s,a) \in \mathbb{R}^{6}\) (four observation numbers, two action numbers) to a
vector \(z(x) \in \mathbb{R}^{d}\), \(d = 128\). It is never trained --- a fixed random embedding.
\item \textbf{The feature} \(\varphi(x)\): what actually enters the bonus. Raw features: \(\varphi = z\).
Unit-norm features: \(\varphi = z / \lVert z\rVert_2\) --- the same vector rescaled to length one, so
only its DIRECTION is kept and its length is thrown away.
\item \textbf{The coverage matrix} \(\Lambda\) (\(d \times d\)). At every update the batch rule sets
\(\Lambda = \frac{1}{n}\Phi^{\top}\Phi + \lambda I\), where \(\Phi\) stacks the \(n{=}256\) minibatch
features (row \(i\) is \(\varphi_i^{\top}\)), so \(\frac{1}{n}\Phi^{\top}\Phi =
\frac{1}{n}\sum_i \varphi_i^{\vphantom{\top}}\varphi_i^{\top}\) is the average outer product of a fresh
\(256\)-sample draw from the replay buffer --- so ``coverage'' approximates the run's visitation
distribution so far, resampled at every update, not the most recent transitions --- and \(\lambda I\)
adds the constant \(\lambda\) to every diagonal entry (the \emph{ridge}).
\item \textbf{Eigenvalues, in plain terms.} A symmetric matrix like \(\Lambda\) has \(d\) perpendicular
directions \(v_1,\dots,v_d\) (eigenvectors) with numbers \(\sigma_1,\dots,\sigma_d\) (eigenvalues),
\(\Lambda v_j = \sigma_j v_j\). Here each one splits as \(\sigma_j = c_j + \lambda\) with
\[
c_j \;=\; \frac{1}{n}\sum_{i=1}^{n} \big(\varphi_i^{\top} v_j\big)^{2},
\]
the average squared alignment of the batch's features with direction \(v_j\) --- literally ``how much
data the batch has in direction \(v_j\)''. A direction the features often point along has large
\(c_j\); a direction no feature touches has \(c_j \approx 0\), leaving \(\sigma_j = \lambda\). So the
ridge \(\lambda\) is a FLOOR: the coverage every direction is credited with before any data is seen.
\item \textbf{The bonus.} \(r(x) = \sqrt{\varphi(x)^{\top}\Lambda^{-1}\varphi(x)}\). Written in the
eigenvector basis,
\[
r(x)^{2} \;=\; \sum_{j=1}^{d} \frac{\big(\varphi(x)^{\top} v_j\big)^{2}}{c_j + \lambda},
\]
each component of the feature is divided by the coverage of its direction: components along well-covered
directions contribute little, components along uncovered directions are divided only by the floor
\(\lambda\) and dominate. That is how the bonus rewards novelty.
\end{itemize}

\paragraph{Hypothesis 1: the effective ridge ratio \(\rho\).}
Two reference cases follow directly from the eigen-form above. A feature pointing entirely along an
\emph{uncovered} direction (\(c_j \approx 0\)) gets
\(r_{\text{novel}} = \lVert\varphi\rVert / \sqrt{\lambda}\); one pointing along a direction with coverage
\(c\) gets \(r_{\text{covered}} = \lVert\varphi\rVert / \sqrt{\lambda + c}\). How big is a TYPICAL
coverage? The trace (the sum of the diagonal) gives it in one line:
\[
\sum_{j=1}^{d} c_j
\;=\; \operatorname{trace}\!\Big(\tfrac{1}{n}\Phi^{\top}\Phi\Big)
\;=\; \tfrac{1}{n}\sum_{i=1}^{n} \lVert\varphi_i\rVert^{2}
\;=\; \overline{\lVert\varphi\rVert^{2}},
\]
i.e.\ the \(d\) coverages sum to the average squared feature length, so their average is
\(\bar\mu = \overline{\lVert\varphi\rVert^{2}} / d\). Taking \(c = \bar\mu\) as the typical coverage,
\[
\frac{r_{\text{novel}}}{r_{\text{covered}}}
= \sqrt{\frac{\lambda + \bar\mu}{\lambda}}
= \sqrt{1 + 1/\rho},
\qquad
\rho \;\equiv\; \frac{\lambda}{\bar\mu} \;=\; \frac{\lambda\, d}{\overline{\lVert\varphi\rVert^{2}}}.
\]
\(\rho\) compares the ridge floor to the typical data coverage. \(\rho \ll 1\): the floor sits far below
the data, novel directions stand out, large contrast. \(\rho \ge 1\): the floor drowns the data, every
direction looks the same, contrast \(\to 1\) --- and \(\beta\) cannot restore the contrast, because it
rescales the informative variation and the constant offset together, leaving their ratio fixed at the
contrast itself. (This is an order-of-magnitude summary --- the
real spectrum is uneven, hence ``typical'' --- which is why every run logs its MEASURED
\(\overline{\lVert z\rVert^{2}}\), \(\overline{\lVert\varphi\rVert^{2}}\), \(\rho\), and eigenvalue
range.) Before substituting numbers, note the two limiting behaviors of the contrast --- they make every
number below transparent. For \(\rho \ll 1\) the \(1\) inside the square root is negligible next to
\(1/\rho\), so
\[
\sqrt{1 + 1/\rho} \;\approx\; \sqrt{1/\rho} \;=\; 1/\sqrt{\rho}
\qquad (\rho \ll 1),
\]
and for \(\rho \gg 1\) the first-order expansion \(\sqrt{1+\varepsilon} \approx 1 + \varepsilon/2\) with
\(\varepsilon = 1/\rho\) gives
\[
\sqrt{1 + 1/\rho} \;\approx\; 1 + \frac{1}{2\rho}
\qquad (\rho \gg 1),
\]
i.e.\ a novelty premium of only \(1/(2\rho)\) above a visited state. Now the three regimes of this run,
each worked through in full:
\begin{itemize}
\item \textbf{Run-2 replica (raw features, \(\lambda = 10^{-6}\)).} The measured raw feature scale is
\(\overline{\lVert z\rVert^{2}} \approx 35\), so
\[
\rho \;=\; \frac{\lambda\, d}{\overline{\lVert\varphi\rVert^{2}}}
\;=\; \frac{10^{-6} \times 128}{35}
\;\approx\; 3.7\times10^{-6},
\qquad
\frac{1}{\rho} \;\approx\; 2.7\times10^{5},
\]
\[
\text{contrast} \;=\; \sqrt{1 + 2.7\times10^{5}} \;\approx\; \sqrt{2.7\times10^{5}}
\;=\; 1/\sqrt{\rho} \;\approx\; 520.
\]
A never-visited direction pays about \(520\) times what a typically visited one pays --- an unmistakable
``go where you have not been'' signal.
\item \textbf{Run-3.1.1, small ridge (unit-norm features, \(\lambda = 10^{-2}\)).} Unit norm forces
\(\overline{\lVert\varphi\rVert^{2}} = 1\), so
\[
\rho \;=\; \frac{10^{-2} \times 128}{1} \;=\; 1.28,
\qquad
\frac{1}{\rho} \;=\; 0.78,
\]
\[
\text{contrast} \;=\; \sqrt{1 + 0.78} \;=\; \sqrt{1.78} \;\approx\; 1.33.
\]
A novel state now earns only \(33\%\) more than a visited one --- a weak preference, easily buried by
reward noise and function-approximation error.
\item \textbf{Run-3.1.1, large ridge (unit-norm features, \(\lambda = 1\)).}
\[
\rho \;=\; \frac{1 \times 128}{1} \;=\; 128,
\qquad
\text{contrast} \;=\; \sqrt{1 + 1/128} \;\approx\; 1 + \frac{1}{2 \times 128} \;=\; 1.004,
\]
using the large-\(\rho\) expansion (the exact value is \(1.0039\)). The novelty premium is \(0.4\%\) ---
invisible under training noise, hence effectively no exploration signal at any \(\beta\).
\end{itemize}
Prediction: reward should track the measured \(\rho\), not the raw/unit label itself; unit-norm cells
with small measured \(\rho\) should approach the replica.

\paragraph{Hypothesis 2: the corner-seeking norm bias.}
The competing account says raw features helped run 2 for a reason unrelated to novelty contrast. The
encoder's bias parameters are initialized to zero and frozen, so the network is a chain of ``multiply by
a matrix'' and ``replace negatives by zero'' steps --- and both steps scale: doubling the input doubles
every intermediate value and hence the output, \(z(\alpha x) = \alpha\, z(x)\) for \(\alpha \ge 0\).
Consequently the raw feature LENGTH carries the input length,
\(\lVert z(x)\rVert = \lVert x\rVert \cdot g(x / \lVert x\rVert)\) with \(g\) depending only on the input
direction, and the raw-feature bonus factorizes as
\[
r(x) \;=\; \lVert z(x)\rVert \cdot \sqrt{u^{\top}\Lambda^{-1} u},
\qquad u = z / \lVert z\rVert :
\]
a novelty part (the square root) TIMES a visitation-independent part
\(\lVert z(x)\rVert \propto \lVert(s,a)\rVert\). What makes \(\lVert(s,a)\rVert\) large? The input is
six-dimensional --- position \((x,y)\) spanning \([-6,6]\times[-4.5,4.5]\) around the arena center,
velocity clipped to \(\pm5\) per axis by the environment, and the action in \([-1,1]^2\) --- so the
maximum squared-norm contributions are position \(56.25\), velocity \(50\), action \(2\): velocity
contributes almost as much as position. At a given speed the norm is largest near the maze corners
(including the \texttt{top\_right} goal corner and equally the \texttt{bottom\_left} start corner), so
``corner-seeking'' is the position shadow of what is really an extreme-state premium: raw features pay a
permanent bonus premium for jointly extreme position AND speed, independent of novelty; unit-norm divides
exactly this factor out. If THIS were the driver, unit-norm cells would stay far below the replica even
at matched measured \(\rho\).

\paragraph{On the clip (literature).}
Neither the original RND paper (Burda et al.\ 2018, \S2.4 --- intrinsic reward divided by a running
standard-deviation estimate of intrinsic returns) nor CleanRL's RND implementation clips the intrinsic
reward; the clip axis therefore measures a non-standard knob, and running-std normalization is the
literature-aligned alternative left to future work.

\paragraph{Findings.}
\begin{enumerate}[label=(\roman*),leftmargin=*]
\item \emph{The replication succeeded (RQ1)}: run-2's exact configuration under the current code reaches
\(\bar{R}=52.76 \pm 6.18\) (\(n{=}33\)), matching run-2's \(44.49 \pm 3.32\) (the difference is \(1.2\)
combined standard errors) --- so the run-3.1.1 gap was caused by the implementation deltas, not by the
environment, the trainer, or the float64 solver.
\item \emph{The clip is the single most damaging factor in the low-\(\rho\) regime (RQ2)}: adding clip
\(=5\) to the replica cell alone collapses \(52.8 \to 4.8\), and the measured saturation explains why ---
\(98\%\) of all computed bonuses hit the cap, erasing the novelty ranking. Across the grid the
relationship is bucket-level (the heavily saturated cells are statistically indistinguishable from one
another): \(89\)--\(98\%\) saturation \(\to\) reward \(5\)--\(8\); \(36\)--\(48\%\) \(\to\)
\(10\)--\(12\); at \(\rho \ge 1\) the clip almost never binds (\(1.6\%\) hits) and is inert
(\(17.2\) vs \(17.2\)) --- exactly the boundedness argument of Section~\ref{sec:train-run-3-1-1}'s
comparison paragraph, and consistent with the literature note that neither the RND paper nor CleanRL
clips the intrinsic reward.
\item \emph{The ridge acts through the measured ratio \(\rho\) (RQ3, Hypothesis 1)}: in \(\rho\) order,
the six no-clip cells score \(52.8\) (\(\rho{=}3.7\times10^{-6}\)), \(26.1\) (\(1.3\times10^{-4}\)),
\(35.2\) (\(3.7\times10^{-4}\)), \(26.5\) (\(1.3\times10^{-2}\)), \(28.0\) (\(3.4\times10^{-2}\)),
\(17.2\) (\(1.28\)). The top cell is the smallest-\(\rho\) cell and the bottom the largest, as
pre-registered, but the middle is flat with one inversion: the \(26.1\) (a unit-norm cell) sits below the
\(35.2\) (a raw cell) despite SMALLER \(\rho\) --- the one comparison leaning toward Hypothesis~2, and it
is within \(1.2\) combined standard errors.
\item \emph{Feature normalization matters mainly by raising the floor of reachable \(\rho\) (RQ3;
Hypothesis 2 dropped on parsimony, not excluded)}: one honesty note first --- the pre-registered
prediction ``unit-norm cells with small measured \(\rho\) approach the replica'' turned out to be
untestable inside this grid, because unit-norm features cannot reach the replica's \(\rho\) at all (their
floor is \(\lambda d = 1.28\times10^{-4}\), \(35\times\) the replica's \(3.7\times10^{-6}\)); only the
weaker nearest-neighbor comparison could be assessed. That comparison shows no systematic unit-below-raw
separation beyond noise (one tie --- \(26.5\) vs \(28.0\) --- and the one \(\approx 1.2\)-combined-SE gap
named in (iii)), so Hypothesis~2 is not needed to explain run-2's advantage. The raw features' measured
scale (\(\overline{\lVert z\rVert^{2}} \approx 35\)) simply gives them a \(35\times\) smaller \(\rho\) at
equal \(\lambda\); under unit norm, matching run-2's contrast would need
\(\lambda \approx 3\times10^{-8}\), below anything in the 3.1.1 grid.
\item \emph{Internal consistency}: the run-3.1.1-like cell (unit, \(10^{-2}\), clip 5) at \(\beta{=}0.01\)
scores \(9.3 \pm 3.7\) here (eval), reproducing run-3.1.1's \(9.03 \pm 3.22\) (train) for the same
configuration. Also as expected, \(\beta^\star\) tends to shift inversely with the bonus scale
(small-\(\rho\) cells pick \(\beta^\star \le 10^{-2}\); at \(\beta{=}0.1\) the tiny-\(\rho\) cells
collapse to \(0\) as the intrinsic term swamps the sparse extrinsic reward).
\end{enumerate}
The Slurm-shape comparison ran alongside: at equal silicon, the 16-tasks\(\times\)1-CPU shape delivered
about \(26\%\) more training steps per CPU-hour than 8\(\times\)2 (per node class \(1.15\times\) to
\(1.51\times\)); the measurements, the core-binding pitfall this uncovered, and the submission-order
lesson are recorded in the run folder's \texttt{analysis.md} and the project Slurm rules.

\paragraph{Follow-up sweep (launched 2026-07-03; results pending).}
Finding~(iv) identified a regime the 3.1.1/3.1.2 grids cannot reach: unit-normalized features at a
replica-level effective ridge ratio, which needs \(\lambda \approx 3\times10^{-8}\). A follow-up sweep
in the same run folder (sweep id timestamp \texttt{2026-07-03-21-22}; catalogued in the folder's
\texttt{data/SWEEPS.md}) runs exactly that regime with one cell a decade below the old grid --- unit
normalization, \(\lambda = 10^{-8}\) (ratio floor \(\rho = \lambda d = 1.28\times10^{-6}\), below the
replica's measured \(3.7\times10^{-6}\)), no clip --- crossed with five \(\beta\) decades
\(\{10^{-4}, 10^{-3}, 10^{-2}, 10^{-1}, 10^{0}\}\) at 50 seeds each (250 runs; every other setting
identical to the grid cells, standalone eval on). It asks:
\begin{enumerate}[label=(\alph*),leftmargin=*]
\item whether unit normalization at a replica-level (here even smaller) \(\rho\) recovers replica-level
eval reward (\(\approx 52.8\)) --- Hypothesis~1 tested where the grid could not test it;
\item whether it instead stays below the raw-feature replica at matched \(\rho\), which would revive the
extreme-state-premium account (Hypothesis~2, dropped on parsimony in finding~(iv));
\item where \(\beta^\star\) sits for this large-contrast bonus (finding~(v): \(\beta^\star\) shifts
inversely with the bonus scale --- hence the wide five-decade \(\beta\) sweep).
\end{enumerate}
The 250 runs execute as one wave (16 open-partition Slurm jobs \(\times\) 16 single-CPU workers, the
shape the comparison above found fastest; the reservation nodes were deliberately left free for other
work). The run folder's \texttt{experiment\_background.md} records the full launch configuration; this
section will be extended with the results.

\subsubsection{Train run 3.2.1: RND optimizer variants (O1 Adam, O2 AdaGrad, O3 SGD-\(1/t\))}
\label{sec:train-run-3-2-1}

\paragraph{Run.}
Train run~3.2.1 is the first live run of the Part~II optimizer design
(Section~\ref{sec:train-run-3-rnd-optimizers}).  It sweeps the three
non-averaged optimizer families --- Method O1 (Adam, the canonical RND
baseline), Method O2 (AdaGrad, cumulative squared-gradient memory), and Method
O3 (SGD with the shifted \(1/t\) schedule) --- each crossed with the two bonus
readouts \(\{\mathcal{B}_{\mathrm{mse}},\mathcal{B}_{\mathrm{l2}}\}\) and the
intrinsic coefficient \(\beta\); the only optimizer-specific swept knobs are
O3's schedule parameters \((\eta_0,t_0)\).  The RND input is
\texttt{rnd\_next\_state} for every run --- the canonical input every external
RND source distills (Table~\ref{tab:opt-used}) and the same input as the
run-3.1.1 reference arm.  This deviates from the staging preference for
\texttt{rnd\_state} (Section~\ref{sec:train-run-3-rnd-optimizers}, chosen there
as run-1's strongest ordinary RND method); \texttt{rnd\_next\_state} equally
admits distance-to-ground-truth logging, so nothing is lost for the winner
rerun.  Every optimizer constant is fixed at its
PyTorch default and no gradient clipping is used
(Tables~\ref{tab:trainrun3-adam-sweep}--\ref{tab:trainrun3-sgd1t-sweep}).
Standalone eval is OFF (run-3.1.1 standard: no deterministic rollout; every
configuration is scored on the training-episode reward --- the windowed mean
over the past \texttt{n\_eval\_episodes} training episodes, snapshotted every
\texttt{eval\_freq} steps); each run writes its own local JSON,
checkpointed at the eval cadence, under the sweep's \texttt{sweep\_id}, and the
queue worker claims configs in id order (seed outermost, per the revised
convention).  The run requires four new \texttt{train.py} switches ---
\texttt{rnd\_optimizer} (\texttt{adam}\,/\,\texttt{adagrad}\,/\,\texttt{sgd1t}),
\texttt{rnd\_bonus\_readout} (\texttt{mse}\,/\,\texttt{l2}),
\texttt{rnd\_sgd\_eta0}, and \texttt{rnd\_sgd\_t0} --- everything else reuses
the run-3.1.1 stack unchanged.  For comparison,
Table~\ref{tab:trainrun3-2-1-fixed} also lists run-3.1.1's
\texttt{rnd\_next\_state} arm (canonical Adam-MSE RND on the next state) as a
reference column, and the analysis plots will draw that arm's best
configuration as a reference line --- taken from its training-episode reward
curve (the same metric logged here), not from a standalone-eval curve.
Because run~3.2.1 uses the same \texttt{rnd\_next\_state} input, O1's
\(\mathcal{B}_{\mathrm{mse}}\) arm has exactly the reference arm's
configuration (Adam at PyTorch defaults, MSE readout, no clipping), so the
seven \(\beta\) values it shares with run~3.1.1 double as a replication of
that arm.

The run answers three questions.
\begin{enumerate}[label=\textbf{RQ\arabic*},leftmargin=*]
\item \textbf{(readout scale)} Does switching Adam's readout from
\(\mathcal{B}_{\mathrm{mse}}\) to \(\mathcal{B}_{\mathrm{l2}}\) alone change
exploration --- is the RND problem mainly a readout-scale problem?
\item \textbf{(count memory)} Does AdaGrad's cumulative squared-gradient memory
improve exploration over constant-gain Adam at matched readout?
\item \textbf{(decreasing step)} Does the shifted \(1/t\) schedule --- the
closest match to the stochastic-approximation averaging argument --- beat both
constant-gain optimizers, and at which \((\eta_0,t_0)\)?
\end{enumerate}

\subsection{Train run 4 (fully-JAX, sample-time reproduction of run 2)}
\label{sec:train-run-4}

\paragraph{Purpose.}
Train run~4 reproduces Train run~2 with the SAC agent and the \texttt{rnd\_\allowbreak state} intrinsic moved
entirely into JAX: \texttt{sbx} provides the XLA-jitted SAC, and a flax/optax RND predictor,
target, and running observation normalization (\texttt{jax\_\allowbreak rnd.JaxRND}) replace the torch RND, so
the per-step intrinsic path contains no torch. The aim is twofold --- confirm the JAX implementation
reproduces run-2's learning curves, and measure the end-to-end speedup. The algorithm is
identical to run~2; the only intended difference is $5\times10^{5}$ training steps (run~2
used $10^{6}$). In particular the intrinsic reward is recomputed at \textbf{sample time} --- on each sampled
gradient-step batch, exactly as run-2's \texttt{VectorIntrinsicReplayBuffer} does, not when a state is first
visited. The SAC subclass \texttt{SACWithIntrinsic} performs this recompute at \texttt{sbx}'s numpy hook
before the jitted update: the buffer stores the extrinsic reward, and $r=r^{\mathrm{ext}}+\beta\,r^{\mathrm{int}}$
is rebuilt on the sampled batch every gradient step with the predictor trained then. Table~\ref{tab:trainrun4-parity}
lists every hyperparameter against run~2; only a few values differ.

\paragraph{Logging.}
Run~4 adopts the revised logging convention (Section~\ref{sec:logging}): besides \texttt{eval\_\allowbreak history}
and \texttt{train\_\allowbreak history}, every run saves \texttt{train\_\allowbreak episode\_\allowbreak
history} --- one record per completed training episode (episode-level, not step-level, to save space) --- so the
full first-to-last training-episode trajectory is retained. The eval-at-final-parameters special case is
removed: the final evaluation uses the same \texttt{n\_\allowbreak eval\_\allowbreak episodes}
($=100$) as every other evaluation. Run~4 logs the reward groups (\texttt{eval\_\allowbreak history} with the
\texttt{visit\_\allowbreak counts/}$*$ coverage block, \texttt{train\_\allowbreak history}, and
\texttt{train\_\allowbreak episode\_\allowbreak history}) but \textbf{not} the oracle-distance group
(\texttt{distance\_\allowbreak history}): \texttt{log\_\allowbreak distance} is off by default, since run~4 is a
reward comparison and run~2's data already carries the distance-to-ground-truth field. Each sweep's JSONs land
under \texttt{data/<sweep\_\allowbreak id>/local/} (the sweep-id convention), so a rerun extends coverage
without colliding with earlier data.

\paragraph{Comparison and finding.}
Figure~\ref{fig:trainrun4-comparison} overlays the eval extrinsic reward over training for run~2
(\textbf{dashed}) and run~4 (\textbf{solid}), one color per algorithm, together with run~4's
training-evaluation curve (a \textbf{dash-dot} line of the same color) --- the mean extrinsic reward over the
past \texttt{n\_\allowbreak eval\_\allowbreak episodes} completed \emph{training} episodes at each evaluation
step. Each line is drawn only up to the largest step reached by at least $30$ seeds.
\textbf{Run~4 does not reproduce run~2.} The curves match up to $\approx2.5\times10^{5}$ steps, after which
run~2 keeps climbing while run~4 grows slowly; at $5\times10^{5}$ steps run~4 reaches only
$0.39$--$0.52\times$ run~2's reward for every algorithm (Table~\ref{tab:trainrun4-result}). The shortfall is
attributable to the \textbf{SAC backend}, not the intrinsic path or the sample-time recompute:
\texttt{gt\_\allowbreak position\_\allowbreak velocity} uses the \emph{identical} torch visit-count bonus with
the same sample-time recompute as run~2 and differs only in the agent (\texttt{sbx}/JAX SAC vs
Stable-Baselines3/torch SAC), yet shows the same gap. So swapping the SAC implementation to \texttt{sbx} is
\emph{not behavior-preserving} on this sparse-reward task --- even with every audited hyperparameter matched
(Table~\ref{tab:trainrun4-parity}) --- and the divergence appears in the late-training exploitation phase,
after the exploration bonus has largely decayed. Run~4 is, however, \textbf{$1.55$--$1.81\times$ faster} per
run at matched step count ($\approx2.9$--$3.1$\,h vs $\approx4.7$--$5.2$\,h for $5\times10^{5}$ steps on $2$
CPUs; below the $\sim2.2\times$ an isolated single-run benchmark suggested, because the sweep packs $8$ workers
per node on slower, more contended hardware). The sweep was stopped at $395/600$ runs once the gap was
established. \textbf{Takeaway:} folding the RND predictor into JAX is correct and faster, but reproducing
run~2 in JAX would require matching \texttt{sbx}'s SAC to Stable-Baselines3's, which this run shows is not
automatic.

\section{Distance to ground truth metrics}
\label{sec:metrics}

Each metric compares a method's bonus field $p$ to the oracle field $g$ over the grid

\subsection{Scale-fit metrics}
\label{sec:dist-scale}

The scale-fit family is invariant to the overall magnitude of $p$: it fits a single
scalar $c$ and reports the residual, $D = \min_{c}\lVert g - c\,p\rVert$.

\paragraph{\texttt{min\_c\_l1\_diff}.}
\begin{align}
D = \min_{c}\lVert g - c\,p\rVert_1 = \min_c\sum_i|g_i - c\,p_i|,
\end{align}
with $c$ found by bounded 1-D numerical search. An exact minimizer is in fact available: $c^\star$ is the weighted median of the ratios $g_i/p_i$ with weights $|p_i|$ (the $L_1$ counterpart of the $L_2$ projection above), so the numerical search is a convenience, not a necessity. If $p\equiv0$, $D = \lVert g\rVert_1$.

\paragraph{\texttt{min\_c\_l2\_diff}.}
Closed-form least-squares scale:
\begin{align}
D = \lVert g - c^\star p\rVert_2, \qquad c^\star = \frac{\langle g,p\rangle}{\langle p,p\rangle}.
\end{align}
\emph{Why this $c^\star$.} It is the minimizer of $\min_c\lVert g-c\,p\rVert_2$: the squared objective $f(c)=\lVert g-c\,p\rVert_2^2=\langle g,g\rangle-2c\,\langle g,p\rangle+c^2\langle p,p\rangle$ is a convex quadratic in the scalar $c$, so its stationary point $f'(c)=-2\langle g,p\rangle+2c\,\langle p,p\rangle=0$---that is, $c^\star=\langle g,p\rangle/\langle p,p\rangle$---is the global minimum (the quadratic opens upward because $\langle p,p\rangle\ge0$). If $p\equiv0$, $D = \lVert g\rVert_2$.

\subsection{Inverse-ratio metrics}
\label{sec:dist-inv}

These ask whether $p$ is a single constant multiple of $g$ \emph{cell by cell}. Form the
element-wise ratio $z = p \oslash g$ (with $g$ regularized so $|g_i|\ge\varepsilon=10^{-8}$,
sign preserved), fit one constant $c$, and report $D = \min_{c}\lVert c\mathbf{1} - z\rVert$.
A small value means $p/g$ is nearly constant across all cells.

\paragraph{\texttt{min\_c\_l1\_inv}.}
The $L_1$ minimizer is the median:
\begin{align}
D = \sum_i|c^\star - z_i|, \qquad c^\star = \mathrm{median}(z).
\end{align}

\paragraph{\texttt{min\_c\_l2\_inv}.}
The $L_2$ minimizer is the mean:
\begin{align}
D = \lVert c^\star\mathbf{1} - z\rVert_2, \qquad c^\star = \bar{z}.
\end{align}
Equivalent to $\sqrt{n}$ times the standard deviation of $z$.

\subsection{Normalized-geometry metrics}
\label{sec:dist-norm}

Rescale both vectors to a common scale, then compare their shape. One uses max-absolute
scaling and an $L_2$ difference; the other compares directions via the angle between unit
vectors.

\paragraph{\texttt{normalized\_l2}.}
\begin{align}
D = \Big\lVert \frac{g}{\max_i|g_i|+\varepsilon} - \frac{p}{\max_i|p_i|+\varepsilon}\Big\rVert_2,
\end{align}
with $\varepsilon=10^{-10}$ (each vector's peak magnitude scaled to $\approx1$).

\paragraph{\texttt{normalized\_angle\_rad}.}
\begin{align}
D = \arccos\!\big(\mathrm{clip}(\langle u_g, u_p\rangle, -1, 1)\big)\in[0,\pi],
\end{align}
where $u_g = g/(\lVert g\rVert_2+\varepsilon)$ and $u_p = p/(\lVert p\rVert_2+\varepsilon)$
are the unit directions ($\varepsilon=10^{-10}$). $0$ = same direction, $\pi$ = opposite.

Each algorithm and the reason all six / none of the metrics apply to it:

\begin{enumerate}[leftmargin=*,itemsep=3pt]
\item \texttt{no\_exploration} --- \emph{all six.} The bonus is identically zero, a constant
field over the grid, so $p$ is the zero vector and every metric is well-defined (the scale-fit
and inverse-ratio metrics reduce to $\lVert g\rVert$ or to $0$).
\item \texttt{gt\_position} --- \emph{all six.} The oracle visit-count bonus keyed on position
is a fixed function of the cell and needs no action, so $p$ is defined on every grid cell.
\item \texttt{gt\_position\_velocity} --- \emph{all six.} This is the ground-truth field $g$
itself; every metric is defined and returns a self-distance ($\approx 0$).
\item \texttt{rnd\_next\_state} --- \emph{all six.} The bonus is the predictor--target error on
the next state; each grid cell serves as its own next state, so the field is evaluable without
actions.
\item \texttt{rnd\_next\_state\_position\_only} --- \emph{all six.} As \texttt{rnd\_next\_state},
but on the position slice of the next state; still a state-only field.
\item \texttt{rnd\_state} --- \emph{all six.} The bonus is the predictor--target error on the
current state, a function of the cell alone.
\item \texttt{rnd\_linear\_next\_state} --- \emph{all six.} Linear-head RND on the next state;
the readout still depends only on the (next) state, so the field is defined per cell.
\item \texttt{rnd\_state\_action} --- \emph{none.} The bonus is a function of $(s,a)$; without
fixing an action there is no single value per grid cell, so it cannot be compared to the
state-indexed oracle field $g$.
\item \texttt{rnd\_state\_action\_next\_state} --- \emph{none.} The input concatenates the
state, the action, and the next state reached by that action, so the bonus is not a function of
the grid cell alone.
\item \texttt{rnd\_elliptical} --- \emph{none.} The elliptical (Mahalanobis) bonus
$\sqrt{\phi(s,a)^\top\Lambda^{-1}\phi(s,a)}$ is computed on state--action features and
accumulates its covariance $\Lambda$ from actions, so it has no action-free per-cell value to
compare against $g$.
\item \texttt{rnd\_\allowbreak elliptical\_\allowbreak global} --- \emph{none.} The same
state--action bonus as \texttt{rnd\_elliptical}, differing only in that the covariance $\Lambda$ is
the global (cumulative) running matrix rather than the per-batch one; it is still action-conditioned,
so it likewise has no action-free per-cell value to compare against $g$.
\end{enumerate}
